    async function refreshImagePoolFromDisk(){
      const rootPath = String(state.imagePool?.rootPath || "").trim();

      if (!rootPath) {
        toast("Choose Image Folder first");
        return;
      }

      try {
        const { ipcRenderer } = require("electron");
        const result = await ipcRenderer.invoke("rescan-image-pool", rootPath);

        if (!result || !result.ok) {
          toast(result?.error || "Image folder rescan failed");
          return;
        }

        // Для неизменившихся файлов сохраняем прежний pool ID.
        // Поэтому блоки и frame slots с poolImageId не теряют привязку.
        const oldIdByMediaUrl = new Map(
          (state.imagePool.items || []).map(item => [
            String(item.src || ""),
            item.id
          ])
        );

        state.imagePool.items = (result.files || []).map(file => {
          // Формат ТОЧНО совпадает с рабочим loadPoolFiles:
          // media:// + полный путь с прямыми слешами.
          const mediaUrl = `media://${String(file.fullPath || "").replace(/\\/g, "/")}`;

          // Используется прежняя логика определения concept/subfolder.
          const relativePath = getRelativePathFromFileLike(file);
          const conceptInfo = resolveConceptInfo(relativePath, file.name);

          return {
            id: oldIdByMediaUrl.get(mediaUrl) || uid("pool"),
            name: file.name,
            originalFileName: file.name,
            relativePath,
            runtimeFileKey: null,
            src: mediaUrl,
            previewSrc: mediaUrl,
            originalSrc: null,
            concept: conceptInfo.concept,
            subfolder: conceptInfo.subfolder,
            lastModified: file.lastModified || 0,
            size: file.size || 0
          };
        });

        // state.pages не изменяется:
        // не сбрасываются страницы, блоки, тексты, позиции и настройки.
        renderAll();
        applyUI();

        toast(`Image folder rescanned: ${state.imagePool.items.length} images`);
      } catch (error) {
        console.error("Image folder rescan error:", error);
        toast("Image folder rescan failed");
      }
    }

      function assignFileToTarget(file, target){

      // PDF IMAGE DOWNSCALE — только для печати, сайт не трогаем
      // ========================================================
      const PDF_DPI = 200;                       // целевой DPI
      const PDF_JPEG_QUALITY = 0.92;             // качество важнее размера
      const PDF_A4_LANDSCAPE_W_PX = Math.round(297 / 25.4 * PDF_DPI); // ширина A4 landscape при 200 DPI
      const PDF_CANVAS_TO_PRINT_SCALE = PDF_A4_LANDSCAPE_W_PX / PAGE_W; // ≈ 2.083

      let __pdfOriginalImageSrcMap = null;

      function loadImageElementForResample(src){
        return new Promise(resolve => {
          const img = new Image();
          img.onload = () => resolve(img);
          img.onerror = () => resolve(null);
          img.src = src;
        });
      }

      function resampleSrcToSize(loadedImg, targetW, targetH, isPng){
        const nw = loadedImg.naturalWidth || loadedImg.width;
        const nh = loadedImg.naturalHeight || loadedImg.height;
        if (!nw || !nh) return null;

        // не растягиваем: если оригинал меньше цели — берём оригинал как есть
        const finalW = Math.max(1, Math.min(nw, Math.round(targetW)));
        const finalH = Math.max(1, Math.min(nh, Math.round(targetH)));

        const canvas = document.createElement("canvas");
        canvas.width = finalW;
        canvas.height = finalH;
        const ctx = canvas.getContext("2d", { alpha: isPng });
        if (ctx && "imageSmoothingEnabled" in ctx) {
          ctx.imageSmoothingEnabled = true;
          ctx.imageSmoothingQuality = "high";
        }
        if (!isPng) {
          ctx.fillStyle = "#ffffff";
          ctx.fillRect(0, 0, finalW, finalH);
        }
        ctx.drawImage(loadedImg, 0, 0, finalW, finalH);

        // PNG (возможна прозрачность) -> PNG, остальное -> JPEG
        return isPng
          ? canvas.toDataURL("image/png")
          : canvas.toDataURL("image/jpeg", PDF_JPEG_QUALITY);
      }

      async function downscaleAllImagesForPdf(){
        __pdfOriginalImageSrcMap = [];
        const imgs = Array.from(document.querySelectorAll(".page-canvas img"));

        for (const imgEl of imgs) {
          const originalSrc = imgEl.getAttribute("src");
          if (!originalSrc) continue;
          if (originalSrc.startsWith("data:image/svg")) continue; // плейсхолдеры не трогаем

          // реальный размер картинки на странице (CSS px канваса)
          const rect = imgEl.getBoundingClientRect();
          // делим на текущий zoom страницы (на экране stage в 1:1, но на всякий случай безопасно)
          const displayW = rect.width;
          const displayH = rect.height;
          if (displayW < 2 || displayH < 2) continue;

          // целевое разрешение под 200 DPI печати
          const targetW = displayW * PDF_CANVAS_TO_PRINT_SCALE;
          const targetH = displayH * PDF_CANVAS_TO_PRINT_SCALE;

          const isPng = /^data:image\/png/i.test(originalSrc) || /\.png(\?|$)/i.test(originalSrc);

          const loaded = await loadImageElementForResample(originalSrc);
          if (!loaded) continue;

          const nw = loaded.naturalWidth || 0;
          // если оригинал уже не больше цели по ширине — нет смысла пережимать
          if (nw && nw <= targetW + 4) continue;

          const newSrc = resampleSrcToSize(loaded, targetW, targetH, isPng);
          if (!newSrc) continue;

          __pdfOriginalImageSrcMap.push({ el: imgEl, src: originalSrc });
          imgEl.setAttribute("src", newSrc);
        }

        // дать браузеру отрисовать новые картинки
        await new Promise(r => requestAnimationFrame(() => requestAnimationFrame(r)));
      }

      function restoreAllImagesAfterPdf(){
        if (!__pdfOriginalImageSrcMap) return;
        __pdfOriginalImageSrcMap.forEach(item => {
          try { item.el.setAttribute("src", item.src); } catch (e) {}
        });
        __pdfOriginalImageSrcMap = null;
      }

      // экспортируем в window, чтобы main.js мог вызвать через IPC
      window.__downscaleAllImagesForPdf = downscaleAllImagesForPdf;
      window.__restoreAllImagesAfterPdf = restoreAllImagesAfterPdf;

      function sanitizeFileName(name){
        const safe = String(name || "image")
          .replace(/[\\/:*?"<>|]+/g, "_")
          .replace(/\s+/g, " ")
          .trim();
        return safe || "image";
      }

      function splitFileName(name){
        const safe = sanitizeFileName(name);
        const idx = safe.lastIndexOf(".");
        if (idx <= 0) return { base: safe, ext: "" };
        return {
          base: safe.slice(0, idx),
          ext: safe.slice(idx)
        };
      }

      function uniqueFileName(name, usedNames){
        const safe = sanitizeFileName(name || "image");
        if (!usedNames.has(safe)) {
          usedNames.add(safe);
          return safe;
        }
        const { base, ext } = splitFileName(safe);
        let i = 2;
        while (true) {
          const candidate = `${base}-${i}${ext}`;
          if (!usedNames.has(candidate)) {
            usedNames.add(candidate);
            return candidate;
          }
          i++;
        }
      }

      function getOriginalFileNameForImageLike(obj){
        if (!obj) return "";
        if (obj.originalFileName) return sanitizeFileName(obj.originalFileName);
        if (obj.poolImageId) {
          const it = getPoolItemById(obj.poolImageId);
          if (it?.originalFileName) return sanitizeFileName(it.originalFileName);
          if (it?.name) return sanitizeFileName(it.name);
        }
        const data = getImageRenderData(obj);
        const src = data?.src || obj.src || "";
        if (/^data:image\/jpeg/i.test(src)) return "image.jpg";
        if (/^data:image\/png/i.test(src)) return "image.png";
        if (/^data:image\/webp/i.test(src)) return "image.webp";
        return "";
      }

      function getExportHrefForBlock(block){
        const links = state.ui.siteLinks || {};

        switch (block.linkTarget) {
          case "email":
            return links.email ? `mailto:${links.email}` : "";
          case "telegram":
            if (!links.telegram) return "";
            return /^https?:\/\//i.test(links.telegram)
              ? links.telegram
              : `https://t.me/${String(links.telegram).replace(/^@/, "")}`;
          case "phone":
            return links.phone ? `tel:${String(links.phone).replace(/\s+/g, "")}` : "";
          case "website":
            return links.website || "";
          case "behance":
            return links.behance || "";
          case "instagram":
            return links.instagram || "";
          case "custom":
            return block.customLink || "";
          default:
            return "";
        }
      }

          script.onerror = () => reject(new Error('Не удалось загрузить html2canvas. Проверьте сеть.'));
          document.head.appendChild(script);

      // Вспомогательная функция для динамической загрузки библиотеки JSZip
      function loadJSZip() {
        return new Promise((resolve, reject) => {
          if (typeof JSZip !== 'undefined') return resolve(JSZip);
          const script = document.createElement('script');
          script.src = 'https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js';
          script.onload = () => resolve(JSZip);
          script.onerror = () => reject(new Error('Не удалось загрузить JSZip. Проверьте сеть.'));
          document.head.appendChild(script);
        });
      }

      // Скачивание любого бинарного файла (включая media:// в Electron, Base64 и сетевые ссылки)
      async function getBinaryData(src) {
        if (!src) return null;
        
        if (src.startsWith('data:')) {
          const parts = src.split(',');
          const base64 = parts[1];
          const raw = atob(base64);
          const rawLength = raw.length;
          const array = new Uint8Array(new ArrayBuffer(rawLength));
          for (let i = 0; i < rawLength; i++) {
            array[i] = raw.charCodeAt(i);
          }
          return array;
        }

        if (typeof require !== 'undefined') {
          try {
            const fs = require('fs');
            let cleanPath = src.replace(/^media:\/\/\/?/, '');
            cleanPath = decodeURIComponent(cleanPath);
            if (cleanPath.startsWith('/') && cleanPath.charAt(2) === ':') {
              cleanPath = cleanPath.substring(1);
            }
            if (fs.existsSync(cleanPath)) {
              return new Uint8Array(fs.readFileSync(cleanPath));
            }
          } catch (fsErr) {
            console.error("FS Error, fallback to fetch:", fsErr);
          }
        }

        const response = await fetch(src);
        const arrayBuffer = await response.arrayBuffer();
        return new Uint8Array(arrayBuffer);
      }

      // Инициализация скачивания готового архива
      function triggerBlobDownload(blob, filename) {
        const a = document.createElement("a");
        a.href = URL.createObjectURL(blob);
        a.download = filename;
        a.click();
        setTimeout(() => URL.revokeObjectURL(a.href), 10000);
      }

      // Обновленный сборщик реестра картинок (универсальный)
      function buildUsedImageRegistry(pages) {
        const usedNames = new Set();
        const entries = [];
        const objectToExportName = new WeakMap();
        const pageUsage = [];

        function registerUsage(pageIndex, pageLabel, imageLike) {
          if (!imageLike) return null;
          const runtimeFile = getRuntimeFileForImageLike(imageLike);
          const originalName = getOriginalFileNameForImageLike(imageLike) || "image.jpg";

          if (objectToExportName.has(imageLike)) {
            const exportName = objectToExportName.get(imageLike);
            let pageEntry = pageUsage.find(p => p.pageIndex === pageIndex);
            if (!pageEntry) {
              pageEntry = { pageIndex, pageLabel, files: [] };
              pageUsage.push(pageEntry);
            }
            if (!pageEntry.files.includes(exportName)) pageEntry.files.push(exportName);
            return exportName;
          }

          const exportName = uniqueFileName(originalName, usedNames);
          objectToExportName.set(imageLike, exportName);

          entries.push({
            exportName,
            originalName,
            file: runtimeFile
          });

          let pageEntry = pageUsage.find(p => p.pageIndex === pageIndex);
          if (!pageEntry) {
            pageEntry = { pageIndex, pageLabel, files: [] };
            pageUsage.push(pageEntry);
          }
          if (!pageEntry.files.includes(exportName)) pageEntry.files.push(exportName);

          return exportName;
        }

        (pages || []).forEach((page, pageIndex) => {
          (page.blocks || []).forEach(block => {
            if (block.kind === "image") {
              registerUsage(pageIndex, page.sourceLabel || `Page ${pageIndex + 1}`, block);
            }
            if (block.kind === "frame") {
              (block.slots || []).forEach(slot => {
                registerUsage(pageIndex, page.sourceLabel || `Page ${pageIndex + 1}`, slot);
              });
            }
          });
        });

        return { entries, objectToExportName, pageUsage };
      }

async function exportSiteToZip() {
        toast("Подготовка экспорта...");
        persistSiteExportUI();

        const pages = state.pages.slice();
        if (!pages.length) return toast("Нет страниц для экспорта!");

        try {
          const jszipInstance = await loadJSZip();
          const zip = new jszipInstance();

          const registry = buildUsedImageRegistry(pages);
          const html = generateFinalHTMLDocument(registry);
          
          // 1. Записываем чистый HTML без редактора
          zip.file("index.html", html);

          // 2. Записываем постраничный текстовый отчет (манифест)
          let manifestText = `ОТЧЕТ ПО СТРАНИЦАМ ПРЕЗЕНТАЦИИ\n`;
          manifestText += `Проект: ${state.ui.siteTitle || "Portfolio"}\n`;
          manifestText += `Дата экспорта: ${new Date().toLocaleString()}\n`;
          manifestText += `Всего страниц: ${pages.length}\n`;
          manifestText += `--------------------------------------------------\n\n`;

          registry.pageUsage.forEach(item => {
            manifestText += `СТРАНИЦА ${item.pageIndex + 1} — "${item.pageLabel}":\n`;
            if (item.files.length > 0) {
              item.files.forEach(file => {
                manifestText += `  └─ Картинка: images/${file}\n`;
              });
            } else {
              manifestText += `  └─ (Нет картинок на странице)\n`;
            }
            manifestText += `\n`;
          });
          zip.file("manifest.txt", manifestText);

          // 3. Записываем картинки в подпапку images/
          const imagesFolder = zip.folder("images");
          let savedCount = 0;

          for (const entry of registry.entries) {
            let binaryData = null;
            const poolItem = state.imagePool.items.find(it => it.originalFileName === entry.originalName || it.name === entry.originalName);
            const src = poolItem ? poolItem.src : "";

            if (entry.file) {
              const arrayBuffer = await entry.file.arrayBuffer();
              binaryData = new Uint8Array(arrayBuffer);
            } else if (src) {
              try {
                binaryData = await getBinaryData(src);
              } catch (e) {
                console.error("Не удалось прочитать " + entry.originalName, e);
              }
            }

            if (binaryData) {
              imagesFolder.file(entry.exportName, binaryData);
              savedCount++;
            }
          }

          toast("Создание архива...");
          const archiveBlob = await zip.generateAsync({ type: "blob" });
          const archiveName = `${slugify(state.ui.siteTitle || "portfolio")}-site.zip`;
          triggerBlobDownload(archiveBlob, archiveName);

          toast(`Сайт успешно собран! Картинок в архиве: ${savedCount}`);
        } catch (err) {
          console.error(err);
          toast(`Ошибка: ${err.message}`);
        }
      }

        function applyTemplateToSelectedPage(){
        const oldPageForTextTransfer = state.pages.find(p => p.id === state.ui.selectedPageId);
        const oldPage = oldPageForTextTransfer ? JSON.parse(JSON.stringify(oldPageForTextTransfer)) : null;

        function transferModifiedTexts(oldP, newP) {
          if (!oldP || !newP) return;
          const memory = oldP.textMemory || {};
          (oldP.blocks || []).forEach(b => {
            if (b.kind === 'text' && b.userModified) {
              if (b.role) {
                memory[b.role] = b.content;
              } else if (b.style && b.style.semanticType) {
                memory['type_' + b.style.semanticType] = b.content;
              }
            }
          });
          (newP.blocks || []).forEach(b => {
            if (b.kind === 'text') {
              let matchedContent = undefined;
              if (b.role && memory[b.role] !== undefined) {
                matchedContent = memory[b.role];
              } else if (b.style && b.style.semanticType && memory['type_' + b.style.semanticType] !== undefined) {
                matchedContent = memory['type_' + b.style.semanticType];
              }
              if (matchedContent !== undefined) {
                b.content = matchedContent;
                b.userModified = true; 
              }
            }
          });
          newP.textMemory = memory;
        }

        function transferImages(oldP, newP) {
          if (!oldP || !newP) return;
          const memory = oldP.imageMemory || [];
          const oldTargets = getAllRandomTargets(oldP, "page");

          oldTargets.forEach((t, i) => {
            if (hasRealAssignedImage(t.target)) {
              memory[i] = {
                src: t.target.src,
                sourceType: t.target.sourceType,
                poolImageId: t.target.poolImageId,
                originalFileName: t.target.originalFileName,
                runtimeFileKey: t.target.runtimeFileKey,
                imageTransform: deepClone(t.target.imageTransform)
              };
            }
          });

          const newTargets = getAllRandomTargets(newP, "page");

          newTargets.forEach((t, i) => {
            if (memory[i]) {
              const saved = memory[i];
              t.target.src = saved.src;
              t.target.sourceType = saved.sourceType;
              t.target.poolImageId = saved.poolImageId;
              t.target.originalFileName = saved.originalFileName;
              t.target.runtimeFileKey = saved.runtimeFileKey;
              t.target.imageTransform = deepClone(saved.imageTransform);
              t.target.isTemplatePlaceholder = false;
            }
          });

          newP.imageMemory = memory;
        }

        const currentPage = getSelectedPage();
        const template = TEMPLATE_INDEX[state.ui.templateId];
        if (!currentPage) return toast("Select page");
        if (!template) return toast("Select template");

        pushHistory();

        const newPage = createTemplatePage(template);
        newPage.id = currentPage.id;
        newPage.sourceLabel = currentPage.sourceLabel || template.label;
        newPage.conceptIndex = currentPage.conceptIndex || 0;
        newPage.conceptSubfolder = normalizeConceptSubfolder(currentPage.conceptSubfolder || "all");

        const index = state.pages.findIndex(p => p.id === currentPage.id);
        if (index >= 0) state.pages[index] = newPage;

        state.ui.selectedPageId = newPage.id;
        state.ui.selectedBlockIds = [];
        state.ui.selectedBlockId = null;
        state.ui.selectedSlotId = null;
        state.ui.editingBlockId = null;

        transferModifiedTexts(oldPage, newPage);
        transferImages(oldPage, newPage);
        autofillPageAfterTemplateApply(newPage);

        renderAll();
        applyUI();
        toast("Template applied to selected page");
      }

function switchTemplateRelative(step = 1) {
  /*
    Next / Previous Template использует тот же filter engine,
    что и основной Select Template dropdown.
  */
  const filteredTemplates =
    getFilteredTemplatesForCurrentCategory(
      state.ui.templateCategory
    );

  if (!filteredTemplates.length) {
    toast(
      window.TL_FILTER_FAV_ONLY
        ? "No favorite templates in this category"
        : "No templates available"
    );
    return;
  }

  const currentIndex = filteredTemplates.findIndex(
    template => template.id === state.ui.templateId
  );

  const nextIndex = currentIndex >= 0
    ? (
        currentIndex +
        step +
        filteredTemplates.length
      ) % filteredTemplates.length
    : 0;

  state.ui.templateId =
    filteredTemplates[nextIndex].id;

  /*
    Category намеренно не меняется.
    All Styles снимает только style filter,
    но не переключает пользователя между категориями.
  */
  syncTemplateSelectorsUI();

  /*
    Сохраняем существующее поведение:
    Next / Previous Template применяет template
    к выбранной странице.
  */
  applyTemplateToSelectedPage();
}

      function csvEscape(value){
        const str = String(value ?? "");
        if (/[",\n;]/.test(str)) return `"${str.replace(/"/g, '""')}"`;
        return str;
      }

      function parseCsv(text){
        const rows = [];
        let row = [];
        let cell = "";
        let i = 0;
        let inQuotes = false;
        while (i < text.length) {
          const ch = text[i];
          const next = text[i + 1];

          if (inQuotes) {
            if (ch === '"' && next === '"') {
              cell += '"';
              i += 2;
              continue;
            }
            if (ch === '"') {
              inQuotes = false;
              i++;
              continue;
            }
            cell += ch;
            i++;
            continue;
          }

          if (ch === '"') {
            inQuotes = true;
            i++;
            continue;
          }

          if (ch === ';') {
            row.push(cell);
            cell = "";
            i++;
            continue;
          }

          if (ch === '\n') {
            row.push(cell);
            rows.push(row);
            row = [];
            cell = "";
            i++;
            continue;
          }

          if (ch === '\r') {
            i++;
            continue;
          }

          cell += ch;
          i++;
        }

        row.push(cell);
        rows.push(row);
        return rows.filter(r => !(r.length === 1 && r[0] === ""));
      }

      function getTextBlocksInOrder(page){
        return (page.blocks || []).filter(b => b.kind === "text");
      }

      function getImageColumnsInOrder(page){
        const out = [];
        (page.blocks || []).forEach(block => {
          if (block.kind === "image") {
            out.push({ kind:"image", block, target:block, key:`image:${block.id}` });
          }
          if (block.kind === "frame") {
            (block.slots || []).forEach((slot, idx) => {
              out.push({ kind:"slot", block, slot, target:slot, key:`slot:${block.id}:${slot.id}`, index:idx });
            });
          }
        });
        return out;
      }

      function getImageExportDescriptor(imageLike){
        const concept = String(Math.max(0, parseInt(parseConceptFromFile(imageLike) || 0, 10))).padStart(2, "0");
        let subfolder = "00";

        if (imageLike?.poolImageId) {
          const it = getPoolItemById(imageLike.poolImageId);
          const sf = String(it?.subfolder || "root");
          subfolder = sf === "root" || sf === "all" ? "00" : sf;
        }

        const originalName = getOriginalFileNameForImageLike(imageLike) || "";
        if (!originalName) return "";
        return `${concept}-${subfolder}-${originalName}`;
      }

      function buildCsvMatrix(){
        const pages = state.pages.slice();
        const maxTextBlocks = pages.reduce((max, page) => Math.max(max, getTextBlocksInOrder(page).length), 0);
        const maxImageCols = pages.reduce((max, page) => Math.max(max, getImageColumnsInOrder(page).length), 0);

        const header = [
          "page_number",
          "template_group",
          "concept",
          "subfolder"
        ];

        for (let i = 0; i < maxTextBlocks; i++) {
          const n = String(i + 1).padStart(2, "0");
          header.push(`text_${n}_role`);
          header.push(`text_${n}_content`);
        }

        for (let i = 0; i < maxImageCols; i++) {
          const n = String(i + 1).padStart(2, "0");
          header.push(`image_${n}`);
        }

        const rows = [header];

        pages.forEach((page, pageIndex) => {
          const row = [
            String(pageIndex + 1),
            getCompactTemplateDisplayName(page),
            String(page.conceptIndex || 0),
            String(page.conceptSubfolder || "all")
          ];

          const textBlocks = getTextBlocksInOrder(page);
          for (let i = 0; i < maxTextBlocks; i++) {
            const block = textBlocks[i];
            row.push(block?.role || "");
            row.push(block ? htmlToTextareaValue(block.content || "") : "");
          }

          const imageCols = getImageColumnsInOrder(page);
          for (let i = 0; i < maxImageCols; i++) {
            const item = imageCols[i];
            row.push(item ? getImageExportDescriptor(item.target) : "");
          }

          rows.push(row);
        });

        return rows;
      }

      function exportCsv(){
        if (!state.pages.length) return toast("No pages");
        const rows = buildCsvMatrix();
        const csv = rows.map(row => row.map(csvEscape).join(";")).join("\n");
        const blob = new Blob([csv], { type:"text/csv;charset=utf-8" });
        const a = document.createElement("a");
        a.href = URL.createObjectURL(blob);
        a.download = (state.ui.currentJsonFileName || "portfolio-project").replace(/\.json$/i, "") + ".csv";
        a.click();
        setTimeout(() => URL.revokeObjectURL(a.href), 10000);
        toast("CSV exported");
      }

      function buildBlankPageFromCsvRow(rowMap, pageNumber){
        const page = createBlankPage();
        page.sourceLabel = `Page ${pageNumber}`;

        const templateGroupValue = String(rowMap.template_group || "").trim();
        const compactMatch = templateGroupValue.match(/^(.*)_ID-\d+$/i);
        page.templateGroup = compactMatch ? compactMatch[1] : (templateGroupValue || "CSV");
        page.templateId = "csv-import";
        page.visualStyle = "cinematic";
        page.conceptIndex = Math.max(0, parseInt(rowMap.concept || "0", 10) || 0);
        page.conceptSubfolder = normalizeConceptSubfolder(rowMap.subfolder || "all");

        const textKeys = Object.keys(rowMap)
          .filter(k => /^text_\d+_content$/i.test(k))
          .sort((a, b) => a.localeCompare(b, undefined, { numeric:true }));

        let y = 80;
        textKeys.forEach((contentKey, index) => {
          const num = contentKey.match(/^text_(\d+)_content$/i)?.[1] || String(index + 1).padStart(2, "0");
          const role = rowMap[`text_${num}_role`] || `text${num}`;
          const textValue = rowMap[contentKey] || "";
          if (!String(textValue).trim()) return;

          const block = makeTextBlock(
            role,
            80,
            y,
            920,
            72,
            textareaValueToHtml(textValue),
            "bodyText",
            {
              fontSize: index === 0 ? 34 : 18,
              fontWeight: index === 0 ? 600 : 400,
              lineHeight: index === 0 ? 1.05 : 1.4,
              semanticType: index === 0 ? "title" : "body",
              color:"#111111"
            }
          );
          page.blocks.push(block);
          y += index === 0 ? 96 : 88;
        });

        return page;
      }

      function importCsvTextOnly(file){
        if (state.pages.length) {
          toast("CSV import only on empty canvas");
          return;
        }
        const reader = new FileReader();
        reader.onload = () => {
          try {
            const rows = parseCsv(String(reader.result || ""));
            if (rows.length < 2) return toast("CSV empty");

            const header = rows[0].map(v => String(v || "").trim());
            const dataRows = rows.slice(1).filter(r => r.some(cell => String(cell || "").trim()));

            if (!dataRows.length) return toast("CSV has no rows");

            pushHistory();

            state.pages = dataRows.map((row, idx) => {
              const rowMap = {};
              header.forEach((key, i) => {
                rowMap[key] = row[i] != null ? row[i] : "";
              });
              return buildBlankPageFromCsvRow(rowMap, idx + 1);
            });

            state.ui.selectedPageId = state.pages[0]?.id || null;
            state.ui.selectedBlockIds = [];
            state.ui.selectedBlockId = null;
            state.ui.selectedSlotId = null;
            state.ui.editingBlockId = null;

            renderAll();
            applyUI();
            toast(`CSV imported: ${state.pages.length} pages`);
          } catch (err) {
            console.error(err);
            toast("CSV import error");
          }
        };
        reader.readAsText(file, "utf-8");
      }

      function getImageLikeConceptAndSubfolder(obj){
        let concept = 0;
        let subfolder = "00";

        if (obj?.poolImageId) {
          const it = getPoolItemById(obj.poolImageId);
          if (it) {
            concept = Math.max(0, parseInt(it.concept || 0, 10) || 0);
            const sf = String(it.subfolder || "root");
            subfolder = sf === "root" || sf === "all" ? "00" : sf;
          }
        } else {
          concept = Math.max(0, parseInt(parseConceptFromFile(obj) || 0, 10) || 0);
        }

        return {
          concept: String(concept).padStart(2, "0"),
          subfolder: String(subfolder || "00").padStart(2, "0")
        };
      }

      function getMimeExtension(file){
        const type = String(file?.type || "").toLowerCase();
        if (type.includes("jpeg")) return ".jpg";
        if (type.includes("png")) return ".png";
        if (type.includes("webp")) return ".webp";
        if (type.includes("gif")) return ".gif";
        if (type.includes("tiff")) return ".tif";
        return "";
      }

      function makePageFolderFileName(imageLike, usedNamesInPage){
        const originalName = getOriginalFileNameForImageLike(imageLike) || "image";
        const { base, ext } = splitFileName(originalName);
        const meta = getImageLikeConceptAndSubfolder(imageLike);
        const safeBase = sanitizeFileName(base || "image");
        const extension = ext || getMimeExtension(getRuntimeFileForImageLike(imageLike)) || ".jpg";

        const primary = `${meta.concept}-${meta.subfolder}-${safeBase}${extension}`;
        if (!usedNamesInPage.has(primary)) {
          usedNamesInPage.add(primary);
          return primary;
        }

        let i = 2;
        while (true) {
          const candidate = `${meta.concept}-${meta.subfolder}-${safeBase}-${i}${extension}`;
          if (!usedNamesInPage.has(candidate)) {
            usedNamesInPage.add(candidate);
            return candidate;
          }
          i++;
        }
      }

      function buildJpegFileName(page){
        // Имя файла: TemplateGroup_ID + короткая дата.
        // getCompactTemplateDisplayName даёт например "Contact / Outro_ID-84"
        const compact = getCompactTemplateDisplayName(page) || "slide";
        const namePart = slugify(compact) || "slide";
        const d = new Date();
        const pad = n => String(n).padStart(2, "0");
        const datePart = d.getFullYear() + "-" + pad(d.getMonth() + 1) + "-" + pad(d.getDate());
        return namePart + "_" + datePart + ".jpg";
      }

      // Постраничный экспорт оригиналов картинок в ZIP по папкам (page-001, page-002...)
      async function exportPageImagesToZip() {
        toast("Подготовка экспорта изображений...");
        const pages = state.pages.slice();
        if (!pages.length) return toast("Нет страниц для экспорта!");

        try {
          const jszipInstance = await loadJSZip();
          const zip = new jszipInstance();
          let savedCount = 0;

          for (let pageIndex = 0; pageIndex < pages.length; pageIndex++) {
            const page = pages[pageIndex];
            const folderName = `page-${String(pageIndex + 1).padStart(3, "0")}`;
            const pageFolder = zip.folder(folderName);
            const usedNamesInPage = new Set();

            const targets = [];
            (page.blocks || []).forEach(block => {
              if (block.kind === "image") targets.push(block);
              if (block.kind === "frame") (block.slots || []).forEach(slot => targets.push(slot));
            });

            for (const imageLike of targets) {
              const poolItem = imageLike.poolImageId ? getPoolItemById(imageLike.poolImageId) : null;
              const src = poolItem ? poolItem.src : imageLike.src;
              const fileObj = getRuntimeFileForImageLike(imageLike);

              let binaryData = null;
              if (fileObj) {
                const arrayBuffer = await fileObj.arrayBuffer();
                binaryData = new Uint8Array(arrayBuffer);
              } else if (src) {
                try {
                  binaryData = await getBinaryData(src);
                } catch (e) {
                  console.error("Ошибка чтения бинарных данных:", e);
                }
              }

              if (binaryData) {
                const exportName = makePageFolderFileName(imageLike, usedNamesInPage);
                pageFolder.file(exportName, binaryData);
                savedCount++;
              }
            }
          }

          if (savedCount === 0) {
            toast("Не найдено изображений для экспорта!");
            return;
          }

          toast("Упаковка картинок в ZIP...");
          const archiveBlob = await zip.generateAsync({ type: "blob" });
          const archiveName = `${slugify(state.ui.siteTitle || "portfolio")}-images-by-pages.zip`;
          triggerBlobDownload(archiveBlob, archiveName);

          toast(`Экспортировано ${savedCount} картинок по папкам в ZIP-архив!`);
        } catch (err) {
          console.error(err);
          toast(`Ошибка экспорта картинок: ${err.message}`);
        }
      }

        const safe = Object.assign(createDefaultTextLibraryState(), deepClone(textLibrary || {}));
        safe.groups = Array.isArray(safe.groups) ? safe.groups : [];
        safe.groups = safe.groups.map(group => ({

            text:String(item && item.text != null ? item.text : "")
          })) : []
        }));
        safe.panelMode = ["expanded","collapsed","hidden"].includes(safe.panelMode) ? safe.panelMode : "expanded";
        safe.panelWidth = clamp(parseInt(safe.panelWidth, 10) || 420, 360, 680);
        safe.collapsedItemIds = Array.isArray(safe.collapsedItemIds) ? safe.collapsedItemIds.map(String) : [];
        if (!safe.groups.length) {
          const fallback = createDefaultTextLibraryState();
          safe.groups = fallback.groups;
          safe.selectedGroupId = fallback.selectedGroupId;
        }
        if (!safe.selectedGroupId || !safe.groups.find(g => g.id === safe.selectedGroupId)) {
          safe.selectedGroupId = safe.groups[0] ? safe.groups[0].id : null;
        }
        return safe;

      function promptTextlibUniqueName(label, existingNames, fallback){
        const names = new Set((existingNames || []).map(v => String(v || "").trim().toLowerCase()));
        const raw = window.prompt(label, fallback || "New Text");
        if (raw == null) return null;
        const name = String(raw).trim();
        if (!name) {
          toast("Name cannot be empty");
          return null;
        }
        if (names.has(name.toLowerCase())) {
          toast("Name must be unique");
          return null;
        }
        return name;
      }

      parseCsv = function(text){
        const source = String(text || "").replace(/^\uFEFF/, "");
        let delimiter = ";";
        const firstLine = source.split(/\r?\n/)[0] || "";
        const semiCount = (firstLine.match(/;/g) || []).length;
        const commaCount = (firstLine.match(/,/g) || []).length;
        if (commaCount > semiCount) delimiter = ",";

        const rows = [];
        let row = [];
        let cell = "";
        let i = 0;
        let inQuotes = false;

        while (i < source.length) {
          const ch = source[i];
          const next = source[i + 1];

          if (inQuotes) {
            if (ch === '"' && next === '"') {
              cell += '"';
              i += 2;
              continue;
            }
            if (ch === '"') {
              inQuotes = false;
              i++;
              continue;
            }
            cell += ch;
            i++;
            continue;
          }

          if (ch === '"') {
            inQuotes = true;
            i++;
            continue;
          }

          if (ch === delimiter) {
            row.push(cell);
            cell = "";
            i++;
            continue;
          }

          if (ch === "\n") {
            row.push(cell);
            rows.push(row);
            row = [];
            cell = "";
            i++;
            continue;
          }

          if (ch === "\r") {
            i++;
            continue;
          }

          cell += ch;
          i++;
        }

        row.push(cell);
        rows.push(row);
        return rows.filter(r => r.some(c => String(c || "").trim()));
      };

      importTextLibraryCsv = function(file){
        if (!file) return;
        const reader = new FileReader();
        reader.onload = () => {
          try {
            const rows = parseCsv(String(reader.result || ""));
            if (!rows.length || rows.length < 2) return toast("Text Library CSV empty");

            const header = rows[0].map(v => String(v || "").trim().toLowerCase());
            const gi = header.indexOf("group_name");
            const ti = header.indexOf("item_title");
            const xi = header.indexOf("item_text");

            if (gi === -1 || ti === -1 || xi === -1) {
              return toast("CSV needs group_name, item_title, item_text");
            }

            const groups = [];
            const groupsMap = new Map();

            rows.slice(1).forEach((row, rowIndex) => {
              const groupName = String(row[gi] || "").trim() || "Default Group";
              const itemTitle = String(row[ti] || "").trim() || ("Item " + (rowIndex + 1));
              const itemText = String(row[xi] || "");

              if (!groupName && !itemTitle && !itemText) return;

              const key = groupName.toLowerCase();
              let group = groupsMap.get(key);

              if (!group) {
                group = { id:uid("tlg"), name:groupName, items:[] };
                groupsMap.set(key, group);
                groups.push(group);
              }

              let title = itemTitle;
              const used = new Set(group.items.map(i => String(i.title || "").toLowerCase()));

              if (used.has(title.toLowerCase())) {
                let n = 2;
                while (used.has((title + " " + n).toLowerCase())) n++;
                title = title + " " + n;
              }

              group.items.push({
                id:uid("tli"),
                title:title,
                text:itemText
              });
            });

            if (!groups.length) return toast("CSV has no items");

            pushHistory();
            state.textLibrary.groups = groups;
            state.textLibrary.selectedGroupId = groups[0].id;
            state.textLibrary.panelMode = "expanded";
            state.textLibrary.collapsedItemIds = [];
            state.textLibrary = normalizeTextLibraryState(state.textLibrary);

            renderTextLibraryPanel();
            applyUI();
            toast("Text Library CSV imported");
          } catch (err) {
            console.error(err);
            toast("Text Library CSV import error");
          }
        };
        reader.readAsText(file, "utf-8");
      };

        const used = new Set((existing || []).map(v => String(v || "").trim().toLowerCase()));
        let name = String(base || "New").trim() || "New";

        let n = 2;
        while (used.has((name + " " + n).toLowerCase())) n++;

      function textlibV4SafeName(value, fallback){
        const name = String(value || "").replace(/\s+/g, " ").trim();
        return name || fallback || "New Text";
      }

      function textlibV4UniqueName(base, existing){
        const used = new Set((existing || []).map(v => String(v || "").trim().toLowerCase()));
        let name = textlibV4SafeName(base, "New Text");
        if (!used.has(name.toLowerCase())) return name;
        let n = 2;
        while (used.has((name + " " + n).toLowerCase())) n++;
        return name + " " + n;
      }
