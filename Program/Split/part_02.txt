<script>
    // Синхронная автозагрузка всех JS-шаблонов из папки templates для Electron
    (function loadCustomTemplatesSync() {
      try {
        if (typeof require !== 'undefined') {
          const fs = require('fs');
          const path = require('path');
          
          // Абсолютный путь к папке templates в корне приложения
          const templatesDir = path.join(__dirname, 'templates');

          if (fs.existsSync(templatesDir)) {
            const files = fs.readdirSync(templatesDir);
            files.forEach(file => {
              if (file.endsWith('.js')) {
                const filePath = path.join(templatesDir, file);
                const code = fs.readFileSync(filePath, 'utf8');
                // Синхронно исполняем код скрипта в контексте window
                window.eval(code);
                console.log(`[Templates] Успешно загружен: ${file}`);
              }
            });
          } else {
            console.warn('[Templates] Предупреждение: Папка "templates" не найдена в корне проекта.');
          }
        }
      } catch (err) {
        console.error('[Templates] Ошибка при автозагрузке шаблонов:', err);
      }
    })();

      function normalizeHotkeysState(hk){
        const safe = Object.assign({}, HOTKEY_DEFAULTS, hk || {});
        const out = {};
        Object.keys(HOTKEY_DEFAULTS).forEach(k => {
          let val = typeof safe[k] === "string" ? safe[k] : "";
          // undo / redo всегда должны иметь дефолтную комбинацию, даже если в сохранённом стейте пусто
          if (!val && (k === "undo" || k === "redo")) val = HOTKEY_DEFAULTS[k] || "";
          out[k] = val;
        });
        return out;
      }

      function hotkeyComboFromEvent(e){
        const parts = [];
        if (e.ctrlKey || e.metaKey) parts.push("Ctrl");
        if (e.shiftKey) parts.push("Shift");
        if (e.altKey) parts.push("Alt");
        let key = e.key;
        if (["Control","Shift","Alt","Meta"].includes(key)) return null;
        if (key === " ") key = "Space";
        if (key.length === 1) key = key.toUpperCase();
        parts.push(key);
        return parts.join("+");
      }

      function inferSemanticType(role, styleName, content){
        const roleNorm = String(role || "").toLowerCase();
        const styleNorm = String(styleName || "").toLowerCase();
        const html = String(content || "").toLowerCase();
        if (styleNorm.includes("overlay") || roleNorm.includes("overlay")) return styleNorm.includes("title") ? "overlayTitle" : "overlayBody";
        if (styleNorm.includes("smallmeta")) return "smallMeta";
        if (styleNorm.includes("statement")) return "statement";
        if (styleNorm.includes("overline") || roleNorm.includes("meta") || roleNorm.includes("kicker")) return "meta";
        if (styleNorm.includes("subtitle") || roleNorm.includes("subtitle") || roleNorm === "sub" || roleNorm === "sub1" || roleNorm === "sub2") return "subtitle";
        if (styleNorm.includes("title") || roleNorm.includes("title") || roleNorm.includes("headline")) return "title";
        if (roleNorm.includes("links") || roleNorm.includes("small")) return "smallMeta";
        if (styleNorm.includes("body") || roleNorm.includes("body") || roleNorm.includes("steps") || html.includes("<br>")) return "body";
        return "body";
      }

      function normalizeConceptNumber(value){
        const n = parseInt(String(value ?? "").trim(), 10);
        return Number.isFinite(n) && n > 0 ? n : 0;
      }

      function normalizeConceptSubfolder(value){
        const str = String(value ?? "").trim().toLowerCase();
        if (str === "all") return "all";
        if (str === "root") return "root";
        if (/^\d{2}$/.test(str)) return str;
        return "all";
      }

      function changeImagePoolConceptBy(step){
        /*
          Реальные номера Concept берём прямо из Image Pool,
          а не из options select.

          Причина: select — это UI выбранной страницы, и он может
          содержать служебное значение 0 либо обновляться только
          после setPageConcept(...).
        */
        const concepts = Array.from(
          new Set(
            (state.imagePool?.items || [])
              .map(item => normalizeConceptNumber(item.concept))
              .filter(number => number > 0)
          )
        ).sort((a, b) => a - b);

        if (!concepts.length) {
          toast("No Concept folders found in Image Pool");
          return;
        }

        const selectedPage = getSelectedPage();

        if (!selectedPage) {
          toast("Select a page first");
          return;
        }

        const currentConcept =
          normalizeConceptNumber(
            selectedPage.conceptIndex
          );

        const currentIndex =
          concepts.indexOf(currentConcept);

        let nextIndex;

        /*
          Если у страницы пока Concept 0:
          → вправо выбирает первый доступный Concept;
          ← влево выбирает последний доступный Concept.
        */
        if (currentIndex < 0) {
          nextIndex = step > 0
            ? 0
            : concepts.length - 1;
        } else {
          nextIndex = clamp(
            currentIndex + step,
            0,
            concepts.length - 1
          );
        }

        const nextConcept = concepts[nextIndex];

        /*
          Уже на первой/последней папке — ничего не меняем.
        */
        if (
          !nextConcept ||
          nextConcept === currentConcept
        ) {
          return;
        }

        /*
          Тот же рабочий путь, что и ручной выбор PAGE CONCEPT:
          history → setPageConcept → autofill / UI.
        */
        pushHistory();

        setPageConcept(
          String(nextConcept),
          { autofill:true }
        );

        /*
          При Concept ±1 и ±10 всегда возвращаемся к All,
          как договорились. Это не изменяет старые страницы:
          меняется только Image Scope активной страницы.
        */
        setPageConceptSubfolder(
          "all",
          { autofill:false }
        );
      }
      function changeImagePoolSubfolderBy(step){
        const select =
          els.pageSubfolderSelect ||
          $("pageSubfolderSelect");

        if (!select) {
          toast("Subfolder selector not found");
          return;
        }

        /*
          Используем порядок, уже заданный в меню:
          All → root → 01 → 02 → 03 ...
        */
        const options = getEnabledSelectOptions(
          select,
          option => {
            const value =
              normalizeConceptSubfolder(option.value);

            return (
              value === "all" ||
              value === "root" ||
              /^\d{2}$/.test(value)
            );
          }
        );

        if (!options.length) {
          toast("No Subfolders found");
          return;
        }

        let currentIndex = options.findIndex(option => {
          return option.value === select.value;
        });

        let nextIndex;

        if (currentIndex < 0) {
          nextIndex = step > 0 ? 0 : options.length - 1;
        } else {
          nextIndex = clamp(
            currentIndex + step,
            0,
            options.length - 1
          );
        }

        const target = options[nextIndex];

        if (!target || target.value === select.value) {
          return;
        }

        select.value = target.value;
        notifyImagePoolSelectChange(select);
      }

      function setImagePoolSubfolderAll(){
        const select =
          els.pageSubfolderSelect ||
          $("pageSubfolderSelect");

        if (!select) return;

        const allOption = Array.from(
          select.options || []
        ).find(option => {
          return (
            !option.disabled &&
            normalizeConceptSubfolder(option.value) === "all"
          );
        });

        if (!allOption || select.value === allOption.value) {
          return;
        }

        select.value = allOption.value;
        notifyImagePoolSelectChange(select);
      }
      function sanitizeRelativePath(path){
        return String(path || "").replace(/\\/g, "/").replace(/^\/+/, "").trim();
      }

      function getRelativePathFromFileLike(fileOrItem){
        if (!fileOrItem) return "";
        const path = sanitizeRelativePath(fileOrItem.relativePath || fileOrItem.webkitRelativePath || "");
        if (path) return path;
        const originalFileName = String(fileOrItem.originalFileName || "").trim();
        const name = String(fileOrItem.name || "").trim();
        return sanitizeRelativePath(originalFileName || name || "");
      }

      function getPathSegments(path){
        return sanitizeRelativePath(path).split("/").filter(Boolean);
      }

      function parseConceptFolderSegment(seg){
        const part = String(seg || "").trim();
        if (!part) return 0;
        if (/^\d+$/.test(part)) return normalizeConceptNumber(part);
        const match = part.match(/^(\d{1,3})(?:[\s_-].*)?$/);
        if (match) return normalizeConceptNumber(match[1]);
        return 0;
      }

      function parseSubfolderSegment(seg){
        const part = String(seg || "").trim().toLowerCase();
        if (!part) return "";
        if (/^\d{2}$/.test(part)) return part;
        const match = part.match(/^(\d{2})(?:[\s_-].*)?$/);
        return match ? match[1] : "";
      }

      function resolveConceptInfo(relativePath, fileName){
        const parts = getPathSegments(relativePath);
        if (!parts.length) {
          const fromName = parseConceptFromName(fileName);
          if (fromName > 0) return { concept:fromName, subfolder:"root" };
          return { concept:0, subfolder:"root" };
        }

        for (let i = 0; i < parts.length; i++) {
          const seg = parts[i];
          const concept = parseConceptFolderSegment(seg);
          if (concept <= 0) continue;

          const next = parts[i + 1] || "";
          const nextSubfolder = parseSubfolderSegment(next);
          const hasMoreAfterNext = i + 2 < parts.length;
          if (hasMoreAfterNext && nextSubfolder) {
            return { concept, subfolder:nextSubfolder };
          }

          if (nextSubfolder && i + 1 < parts.length - 1) {
            return { concept, subfolder:nextSubfolder };
          }

          return { concept, subfolder:"root" };
        }

        const fromName = parseConceptFromName(fileName);
        if (fromName > 0) return { concept:fromName, subfolder:"root" };
        return { concept:0, subfolder:"root" };
      }

      function parseConceptFromFolderPath(path){
        return resolveConceptInfo(path, "").concept;
      }

      function parseConceptSubfolderFromFolderPath(path){
        return resolveConceptInfo(path, "").subfolder;
      }

      function parseConceptFromName(name){
        const fileName = String(name || "").split("/").pop().split("\\").pop().trim();
        if (!fileName) return 0;
        const baseName = fileName.replace(/\.[^.]+$/, "");
        const match = baseName.match(/^(\d+)(?:$|[^0-9])/);
        return match ? normalizeConceptNumber(match[1]) : 0;
      }

      function resolveConcept(relativePath, fileName){
        return resolveConceptInfo(relativePath, fileName).concept;
      }

      function parseConceptFromFile(fileOrItem){
        const relativePath = getRelativePathFromFileLike(fileOrItem);
        const fileName = String(fileOrItem?.originalFileName || fileOrItem?.name || "").trim();
        return resolveConcept(relativePath, fileName);
      }

      function parseConceptSubfolderFromFile(fileOrItem){
        const relativePath = getRelativePathFromFileLike(fileOrItem);
        const fileName = String(fileOrItem?.originalFileName || fileOrItem?.name || "").trim();
        return resolveConceptInfo(relativePath, fileName).subfolder;
      }
