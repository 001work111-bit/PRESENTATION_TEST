function approvedSplitExportNormalizeText(value){
  return String(value || "")
    .replace(/\s+/g, " ")
    .trim()
    .toLowerCase();
}

function approvedSplitExportFindButton(aliases){
  const root = document.getElementById("approvedDockRoot");

  if (!root) return null;

  const normalizedAliases = aliases.map(approvedSplitExportNormalizeText);

  const buttons = Array.from(
    root.querySelectorAll("button, input[type='button'], input[type='submit']")
  );

  return buttons.find((button) => {
    /*
      Не ищем кнопку повторно внутри уже собранного Export runtime.
    */
    if (button.closest("#approvedSplitExportRuntime")) {
      return false;
    }

    const rawText =
      button.tagName === "INPUT"
        ? button.value
        : button.textContent;

    const buttonText = approvedSplitExportNormalizeText(rawText);

    return normalizedAliases.some((alias) => buttonText === alias);
  }) || null;
}

function approvedSplitExportCreateCard(title){
  const card = document.createElement("section");
  card.className = "approved-export-card";

  const heading = document.createElement("h3");
  heading.className = "approved-export-title";
  heading.textContent = title;

  const grid = document.createElement("div");
  grid.className = "approved-export-grid";

  card.append(heading, grid);

  return { card, grid };
}

function approvedSplitExportAppendButton(grid, button, wide){
  if (!button) return false;

  if (wide) {
    button.classList.add("approved-export-wide");
  } else {
    button.classList.remove("approved-export-wide");
  }

  /*
    appendChild перемещает настоящий элемент.
    Его ID, onclick, addEventListener и внутреннее состояние сохраняются.
  */
  grid.appendChild(button);
  return true;
}

(function initApprovedSplitPanelResizeV4(){
  const STORAGE_KEY = "approvedSplitPanelHeightsV4";
  const MIN_HEIGHT = 220;
  const VIEWPORT_BOTTOM_GAP = 24;
  const HANDLE_ID = "approvedSplitOuterResizeHandle";

  let resizeState = null;
  let initialized = false;

  function getPanel(){
    return document.querySelector("body.shell-split-active .left-panel");
  }

  function getSectionSelect(){
    return document.getElementById("approvedSplitSectionSelect");
  }

  function getCurrentSection(){
    const select = getSectionSelect();

    return (
      select?.value ||
      document.body.dataset.approvedSplitSection ||
      "default"
    );
  }

  function isResizableSplitPanel(panel){
    if (!document.body.classList.contains("shell-split-active")) {
      return false;
    }

    if (!panel) {
      return false;
    }

    const style = window.getComputedStyle(panel);

    return (
      style.display !== "none" &&
      style.visibility !== "hidden" &&
      panel.getBoundingClientRect().width > 0
    );
  }

  function readHeights(){
    try {
      const raw = localStorage.getItem(STORAGE_KEY);

      if (!raw) {
        return {};
      }

      const value = JSON.parse(raw);

      return value && typeof value === "object" ? value : {};
    } catch (error) {
      return {};
    }
  }

  function saveHeights(heights){
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(heights));
    } catch (error) {
      /* localStorage can be unavailable. */
    }
  }

  /*
    Максимум считается от настоящего верха ВНЕШНЕЙ панели.
    Поэтому Concept / Load Deck могут расти вниз до края окна.
  */
  function getMaximumHeight(panel){
    if (!panel) {
      return MIN_HEIGHT;
    }

    const top = panel.getBoundingClientRect().top;

    return Math.max(
      MIN_HEIGHT,
      Math.floor(window.innerHeight - top - VIEWPORT_BOTTOM_GAP)
    );
  }

  function clampHeight(panel, value){
    const number = Number(value) || MIN_HEIGHT;

    return Math.max(
      MIN_HEIGHT,
      Math.min(getMaximumHeight(panel), Math.round(number))
    );
  }

  function getDefaultHeight(panel){
    const actualHeight = Math.round(panel.getBoundingClientRect().height);

    if (actualHeight >= MIN_HEIGHT) {
      return clampHeight(panel, actualHeight);
    }

    return Math.min(380, getMaximumHeight(panel));
  }

  function setPanelHeight(panel, value){
    if (!panel) {
      return;
    }

    const height = clampHeight(panel, value);
    const maxHeight = getMaximumHeight(panel);

    panel.style.setProperty(
      "--approved-split-outer-height",
      `${height}px`
    );

    panel.style.setProperty(
      "--approved-split-outer-max-height",
      `${maxHeight}px`
    );

    panel.style.height = `${height}px`;
    panel.style.maxHeight = `${maxHeight}px`;

    updateHandle();
  }

  function applyCurrentSectionHeight(){
    const panel = getPanel();

    if (!isResizableSplitPanel(panel)) {
      updateHandle();
      return;
    }

    const heights = readHeights();
    const section = getCurrentSection();
    const stored = heights[section];

    setPanelHeight(panel, stored || getDefaultHeight(panel));
  }

  function persistCurrentHeight(){
    const panel = getPanel();

    if (!panel || !isResizableSplitPanel(panel)) {
      return;
    }

    const heights = readHeights();

    heights[getCurrentSection()] = clampHeight(
      panel,
      panel.getBoundingClientRect().height
    );

    saveHeights(heights);
  }

  function getHandle(){
    let handle = document.getElementById(HANDLE_ID);

    if (handle) {
      return handle;
    }

    handle = document.createElement("div");
    handle.id = HANDLE_ID;
    handle.className = "approved-split-outer-resize-handle";
    handle.setAttribute("role", "separator");
    handle.setAttribute("aria-label", "Resize Split panel");
    handle.setAttribute("aria-orientation", "horizontal");

    document.body.appendChild(handle);

    handle.addEventListener("pointerdown", startResize);

    return handle;
  }

  function updateHandle(){
    const handle = getHandle();
    const panel = getPanel();

    if (!isResizableSplitPanel(panel) || resizeState?.panel !== panel) {
      if (!isResizableSplitPanel(panel)) {
        handle.classList.remove("is-visible");
        return;
      }
    }

    if (!isResizableSplitPanel(panel)) {
      handle.classList.remove("is-visible");
      return;
    }

    const rect = panel.getBoundingClientRect();

    handle.style.left = `${Math.round(rect.left + 10)}px`;
    handle.style.width = `${Math.max(0, Math.round(rect.width - 20))}px`;

    /*
      Handle расположен СНАРУЖИ:
      начинается на внешней нижней границе и уходит вниз.
      Он не находится в content-area панели.
    */
    handle.style.top = `${Math.round(rect.bottom - 1)}px`;
    handle.classList.add("is-visible");
  }

  function startResize(event){
    const panel = getPanel();

    if (
      event.button !== 0 ||
      resizeState ||
      !isResizableSplitPanel(panel)
    ) {
      return;
    }

    event.preventDefault();
    event.stopPropagation();

    resizeState = {
      pointerId: event.pointerId,
      panel,
      startY: event.clientY,
      startHeight: panel.getBoundingClientRect().height
    };

    document.body.classList.add("approved-split-panel-is-resizing");

    if (event.currentTarget.setPointerCapture) {
      event.currentTarget.setPointerCapture(event.pointerId);
    }

    window.addEventListener("pointermove", moveResize, true);
    window.addEventListener("pointerup", endResize, true);
    window.addEventListener("pointercancel", endResize, true);
  }

  function moveResize(event){
    if (!resizeState) {
      return;
    }

    event.preventDefault();

    const nextHeight =
      resizeState.startHeight +
      (event.clientY - resizeState.startY);

    setPanelHeight(resizeState.panel, nextHeight);
  }

  function endResize(){
    if (!resizeState) {
      return;
    }

    persistCurrentHeight();

    resizeState = null;

    document.body.classList.remove("approved-split-panel-is-resizing");

    window.removeEventListener("pointermove", moveResize, true);
    window.removeEventListener("pointerup", endResize, true);
    window.removeEventListener("pointercancel", endResize, true);

    updateHandle();
  }

  function bindSectionSelect(){
  const select = getSectionSelect();

  if (!select || select.dataset.resizeV4Bound === "1") {
    return;
  }

  select.dataset.resizeV4Bound = "1";

  select.addEventListener("change", function(){
    /*
      Раньше здесь были три setTimeout: 0 / 80 / 250.
      Из-за них панель несколько раз меняла высоту и визуально
      «достраивалась» после переключения раздела.

      Теперь высота применяется один раз на ближайшем render-frame,
      после того как setSplitSection() уже поменял section/class state.
    */
    requestAnimationFrame(function(){
      applyCurrentSectionHeight();
      updateHandle();
    });
  });
}

  function install(){
    getHandle();
    bindSectionSelect();
    applyCurrentSectionHeight();
    updateHandle();
  }

  function boot(){
    if (!initialized) {
      initialized = true;

      window.addEventListener("resize", function(){
        applyCurrentSectionHeight();
        updateHandle();
      });

      /*
        При переключении Split / AIPC / Pages layout может меняться
        без resize окна.
      */
      new MutationObserver(function(){
        updateHandle();
      }).observe(document.body, {
        attributes:true,
        attributeFilter:["class"]
      });
    }

    install();

    setTimeout(install, 300);
    setTimeout(install, 1000);
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", boot, { once:true });
  } else {
    boot();
  }
})();

function onItemClick(id, e){
          if (e.shiftKey && ppAnchorId) {
            const a = state.pages.findIndex(p => p.id === ppAnchorId);
            const b = state.pages.findIndex(p => p.id === id);
            if (a >= 0 && b >= 0) {
              const lo = Math.min(a, b), hi = Math.max(a, b);
              ppSelectedIds = state.pages.slice(lo, hi + 1).map(p => p.id);
            }
          } else if (e.ctrlKey || e.metaKey) {
            if (ppSelectedIds.includes(id)) ppSelectedIds = ppSelectedIds.filter(x => x !== id);
            else ppSelectedIds = [...ppSelectedIds, id];
            ppAnchorId = id;
          } else {
            ppSelectedIds = [id];

        function getSelectedIndicesSorted(){

        function moveSelectedUp(){
          const indices = getSelectedIndicesSorted();
          if (!indices.length || indices[0] === 0) return;
          pushHistory();
          indices.forEach(i => {
            const tmp = state.pages[i - 1];
            state.pages[i - 1] = state.pages[i];
            state.pages[i] = tmp;
          });

          applyUI();

        function moveSelectedDown(){
          const indices = getSelectedIndicesSorted();
          if (!indices.length || indices[indices.length - 1] === state.pages.length - 1) return;
          pushHistory();
          for (let k = indices.length - 1; k >= 0; k--){
            const i = indices[k];
            const tmp = state.pages[i + 1];
            state.pages[i + 1] = state.pages[i];
            state.pages[i] = tmp;
          }
          renderAll();
          applyUI();
          renderPanel();
        }

        function duplicateSelected(){
          const indices = getSelectedIndicesSorted();
          if (!indices.length) return;
          pushHistory();
          // дублируем с конца, чтобы индексы не сбивались
          const newIds = [];
          for (let k = indices.length - 1; k >= 0; k--){
            const i = indices[k];
            const copy = reIdPage(state.pages[i]);
            state.pages.splice(i + 1, 0, copy);
            newIds.push(copy.id);
          }
          ppSelectedIds = newIds;
          ppAnchorId = newIds[0] || null;
          state.ui.selectedPageId = newIds[0] || state.ui.selectedPageId;
          renderAll();
          applyUI();
          renderPanel();

    function formatPresentationMetaText(text) {
    if (typeof text !== 'string') return text;
    let parts = [];
    if (text.includes('|')) {
      parts = text.split('|').map(p => p.trim());
    } else if (text.includes('\n')) {
      parts = text.split('\n').map(p => p.trim());
    } else {
      parts = [text.trim()];
    }
    let formattedParts = [];
    parts.forEach(part => {
      if (!part) return;
      const eqIndex = part.indexOf('=');
      if (eqIndex !== -1) {
        const keyName = part.substring(0, eqIndex).trim();
        const val = part.substring(eqIndex + 1).trim();
        formattedParts.push(`${keyName}:\n${val}`);
      } else {
        formattedParts.push(part);
      }
    });
    return formattedParts.join('\n\n');
  }

    if (!Array.isArray(plan.pages)) {
      throw new Error('В AI JSON отсутствует массив "pages".');
    }

    return plan;
  }

  function aiDeckGetTemplateImageSlotCount(templateId){
    const template = TEMPLATE_INDEX[templateId];

    if (!template) return 0;

    const samplePage = createTemplatePage(template);

    return (samplePage.blocks || []).reduce((total, block) => {
      if (block.kind === "image") return total + 1;

      if (block.kind === "frame") {
        return total + (
          (block.slots || []).length ||
          getFrameLayoutCount(block.layout || "2x2")
        );
      }

      return total;
    }, 0);
  }

  function aiDeckGetTemplateTextRoles(templateId){
    const template = TEMPLATE_INDEX[templateId];

    if (!template) return [];

    const samplePage = createTemplatePage(template);

    return (samplePage.blocks || [])
      .filter(block => block.kind === "text")
      .map(block => String(block.role || "").trim())
      .filter(Boolean);
  }

  function aiDeckGetImageCountForAssignment(conceptIndex, conceptSubfolder){
    const concept = normalizeConceptNumber(conceptIndex || 0);
    const subfolder = normalizeConceptSubfolder(conceptSubfolder || "all");
    const pool = state.imagePool?.items || [];

    if (concept === 0) {
      return pool.length;
    }

    if (subfolder === "all") {
      return pool.filter(item =>
        normalizeConceptNumber(item.concept || 0) === concept
      ).length;
    }

    return pool.filter(item =>
      normalizeConceptNumber(item.concept || 0) === concept &&
      normalizeConceptSubfolder(item.subfolder || "root") === subfolder
    ).length;
  }

  function aiDeckHasConceptAssignment(conceptIndex, conceptSubfolder){
    return aiDeckGetImageCountForAssignment(
      conceptIndex,
      conceptSubfolder
    ) > 0;
  }

  function aiDeckNormalizeImageFill(page){
    const source = (
      page?.imageFill &&
      typeof page.imageFill === "object" &&
      !Array.isArray(page.imageFill)
    )
      ? page.imageFill
      : {};

    return {
      mode:["auto", "none", "manual"].includes(source.mode)
        ? source.mode
        : "auto",

      strategy:["hero-first", "all-slots", "manual"].includes(source.strategy)
        ? source.strategy
        : "hero-first",

      noRepeats:source.noRepeats == null
        ? !!state.ui.noRepeats
        : !!source.noRepeats
    };
  }

  function aiDeckValidatePlan(plan){
    const errors = [];
    const warnings = [];

    const validTemplateIds = new Set(Object.keys(TEMPLATE_INDEX));
    const knownTextRoles = new Set([
      "kicker",
      "overline",
      "meta",
      "meta1",
      "meta2",
      "title",
      "overlay-title",
      "subtitle",
      "sub",
      "sub1",
      "sub2",
      "body",
      "body-right",
      "statement",
      "steps",
      "services",
      "links",
      "small",
      "smallMeta",
      "values",
      "value1"
    ]);

    const globalNoRepeatUsage = new Map();

    if (plan.schemaVersion !== "ai-deck-plan-v1") {
      errors.push(
        'schemaVersion должен быть строго равен "ai-deck-plan-v1".'
      );
    }

    if (!plan.pages.length) {
      errors.push('Массив "pages" пуст.');
    }

    if (
      plan.document?.visualStyle &&
      !["cinematic", "editorial-blue"].includes(plan.document.visualStyle)
    ) {
      errors.push(
        `document.visualStyle "${plan.document.visualStyle}" не поддерживается.`
      );
    }

    plan.pages.forEach((page, index) => {
      const pageNo = index + 1;

      if (!page || typeof page !== "object" || Array.isArray(page)) {
        errors.push(`Страница ${pageNo}: должна быть объектом.`);
        return;
      }

      if (!page.templateId || typeof page.templateId !== "string") {
        errors.push(`Страница ${pageNo}: отсутствует templateId.`);
        return;
      }

      if (!validTemplateIds.has(page.templateId)) {
        errors.push(
          `Страница ${pageNo}: templateId "${page.templateId}" не существует в редакторе.`
        );
        return;
      }

      const template = TEMPLATE_INDEX[page.templateId];
      const conceptIndex = normalizeConceptNumber(page.conceptIndex || 0);
      const conceptSubfolder = normalizeConceptSubfolder(
        page.conceptSubfolder || "all"
      );

      if (
        page.conceptIndex != null &&
        !Number.isFinite(Number(page.conceptIndex))
      ) {
        errors.push(`Страница ${pageNo}: conceptIndex должен быть числом.`);
      }

      if (
        page.conceptSubfolder != null &&
        !["all", "root"].includes(String(page.conceptSubfolder).toLowerCase()) &&
        !/^\d{2}$/.test(String(page.conceptSubfolder))
      ) {
        errors.push(
          `Страница ${pageNo}: недопустимый conceptSubfolder "${page.conceptSubfolder}".`
        );
      }

      if (
        page.text != null &&
        (typeof page.text !== "object" || Array.isArray(page.text))
      ) {
        errors.push(`Страница ${pageNo}: поле text должно быть объектом.`);
      }

      const allowedTextRoles = new Set(
        aiDeckGetTemplateTextRoles(page.templateId)
      );

      if (page.text && typeof page.text === "object") {
        Object.entries(page.text).forEach(([role, content]) => {
          if (!knownTextRoles.has(role)) {
            errors.push(
              `Страница ${pageNo}: роль текста "${role}" не поддерживается редактором.`
            );
            return;
          }

          if (!allowedTextRoles.has(role)) {
            errors.push(
              `Страница ${pageNo}: роль "${role}" отсутствует в шаблоне "${page.templateId}".`
            );
          }

          if (
            typeof content !== "string" &&
            typeof content !== "number"
          ) {
            errors.push(
              `Страница ${pageNo}: текст роли "${role}" должен быть строкой.`
            );
          }
        });
      }

      const imageFill = aiDeckNormalizeImageFill(page);
      const imageSlots = aiDeckGetTemplateImageSlotCount(page.templateId);

      if (imageFill.mode === "manual") {
        warnings.push(
          `Страница ${pageNo}: imageFill.mode "manual" не использует filenames. Картинки останутся пустыми.`
        );
      }

      if (imageFill.mode === "auto" && imageSlots > 0) {
        const availableImages = aiDeckGetImageCountForAssignment(
          conceptIndex,
          conceptSubfolder
        );

        if (!availableImages) {
          errors.push(
            `Страница ${pageNo}: в Image Pool нет картинок для concept ${conceptIndex} / ${conceptSubfolder}.`
          );
        }

        if (imageFill.noRepeats) {
          const usageKey = `${conceptIndex}:${conceptSubfolder}`;
          const currentUsage = globalNoRepeatUsage.get(usageKey) || 0;
          const totalUsage = currentUsage + imageSlots;

          globalNoRepeatUsage.set(usageKey, totalUsage);

          if (totalUsage > availableImages) {
            errors.push(
              `Страница ${pageNo}: для concept ${conceptIndex} / ${conceptSubfolder} недостаточно изображений. ` +
              `Нужно ${totalUsage}, доступно ${availableImages}, потому что включён noRepeats.`
            );
          }
        }
      }

      if (
        plan.document?.visualStyle &&
        template.visualStyle &&
        template.visualStyle !== plan.document.visualStyle
      ) {
        warnings.push(
          `Страница ${pageNo}: template "${page.templateId}" имеет стиль "${template.visualStyle}", ` +
          `а document.visualStyle указан как "${plan.document.visualStyle}".`
        );
      }
    });

    return {
      valid:errors.length === 0,
      errors,
      warnings,
      pagesCount:plan.pages.length
    };
  }

  function aiDeckRenderPreview(plan, validation){
    if (!aiEls.preview) return;

    if (!plan?.pages?.length) {
      aiEls.preview.innerHTML = `
        <div class="ai-deck-preview-empty">
          Нет страниц для preview.
        </div>
      `;
      return;
    }

    aiEls.preview.innerHTML = plan.pages.map((page, index) => {
      const template = TEMPLATE_INDEX[page.templateId];

      const title =
        page.label ||
        page.text?.title ||
        page.text?.["overlay-title"] ||
        template?.label ||
        page.templateId ||
        `Page ${index + 1}`;

      const concept = normalizeConceptNumber(page.conceptIndex || 0);
      const subfolder = normalizeConceptSubfolder(
        page.conceptSubfolder || "all"
      );

      const imageFill = aiDeckNormalizeImageFill(page);
      const imageSlots = aiDeckGetTemplateImageSlotCount(page.templateId);

      return `
        <div class="ai-deck-plan-item">
          <div class="ai-deck-plan-index">${index + 1}</div>
          <div>
            <div class="ai-deck-plan-name">${escapeHtml(String(title))}</div>
            <div class="ai-deck-plan-meta">
              ${escapeHtml(page.templateId || "no-template")} ·
              ${escapeHtml(template?.visualStyle || "unknown")} ·
              concept ${escapeHtml(String(concept))} /
              ${escapeHtml(String(subfolder))} ·
              ${imageSlots} image slot(s) ·
              ${escapeHtml(imageFill.mode)}
            </div>
          </div>
        </div>
      `;
    }).join("");

    if (validation?.errors?.length) {
      aiEls.preview.insertAdjacentHTML(
        "afterbegin",
        `
          <div class="ai-deck-preview-empty"
            style="margin-bottom:10px;color:#ffaaa5;border-color:rgba(255,107,107,.35);">
            План содержит ошибок: ${validation.errors.length}
          </div>
        `
      );
    }
  }

  function aiDeckValidateAndPreview(mode){
    try {
      const plan = aiDeckParseInput();
      const validation = aiDeckValidatePlan(plan);

      lastAiDeckPlan = plan;
      aiDeckRenderPreview(plan, validation);

      if (!validation.valid) {
        aiDeckSetStatus(
          `Сборка остановлена. Ошибок: ${validation.errors.length}\n\n` +
          validation.errors.map(error => `• ${error}`).join("\n"),
          "error"
        );

        return { plan, validation };
      }

      const warnings = validation.warnings.length
        ? `\n\nПредупреждения:\n${validation.warnings.map(w => `• ${w}`).join("\n")}`
        : "";

      aiDeckSetStatus(
        `${mode || "Validation"} complete.\n` +
        `Страниц в плане: ${validation.pagesCount}.${warnings}`,
        "success"
      );

      return { plan, validation };
    } catch (error) {
      lastAiDeckPlan = null;
      aiDeckRenderPreview(null, null);
      aiDeckSetStatus(error.message || "AI JSON error.", "error");
      return null;
    }
  }

  function aiDeckClearTemplateImages(page){
    (page.blocks || []).forEach(block => {
      if (block.kind === "image") {
        clearImageLikeAssignment(block);
      }

      if (block.kind === "frame") {
        (block.slots || []).forEach(slot => {
          clearImageLikeAssignment(slot);
        });
      }
    });
  }

  function aiDeckApplyTextToPage(page, text){
    if (!text || typeof text !== "object") return;

    Object.entries(text).forEach(([role, value]) => {
      (page.blocks || [])
        .filter(block => block.kind === "text" && block.role === role)
        .forEach(block => {
          block.content = textareaValueToHtml(String(value ?? ""));
          block.userModified = true;
        });
    });
  }

  function aiDeckCompilePage(pagePlan, usedImageIds){
    const template = TEMPLATE_INDEX[pagePlan.templateId];

    if (!template) {
      throw new Error(`Template "${pagePlan.templateId}" не найден.`);
    }

    const page = createTemplatePage(template);
    const conceptIndex = normalizeConceptNumber(pagePlan.conceptIndex || 0);
    const conceptSubfolder = normalizeConceptSubfolder(
      pagePlan.conceptSubfolder || "all"
    );

    page.sourceLabel = String(pagePlan.label || template.label || "AI Page");
    page.conceptIndex = conceptIndex;
    page.conceptSubfolder = conceptSubfolder;

    aiDeckApplyTextToPage(page, pagePlan.text || {});

    const imageFill = aiDeckNormalizeImageFill(pagePlan);

    if (imageFill.mode === "none" || imageFill.mode === "manual") {
      aiDeckClearTemplateImages(page);
    }

    if (imageFill.mode === "auto") {
      const targets = [];

      (page.blocks || []).forEach(block => {
        if (block.kind === "image") {
          targets.push({ block, target:block });
        }

        if (block.kind === "frame") {
          (block.slots || []).forEach(slot => {
            targets.push({ block, slot, target:slot });
          });
        }
      });

      if (targets.length) {
        const result = randomizeTargets(page, targets, {
          usedGlobal:usedImageIds,
          strictNoRepeats:imageFill.noRepeats,
          leaveUnfilledOnExhausted:true
        });

        if (!result.changed && result.count === 0) {
          throw new Error(
            `Не удалось заполнить картинками страницу "${page.sourceLabel}".`
          );
        }
      }
    }

    // FIX_AI_EDIT_LOCK_V1:
    // Нормализуем ТОЛЬКО собранную страницу, не мутируя живой state.ui/document/textLibrary.
    // Это чинит потерю inline-редактирования текста после AI Deck Builder.
    ensureStateShape({
      ui: deepClone(state.ui),
      pages:[page],
      document: deepClone(state.document),
      imagePool: state.imagePool,
      textLibrary: deepClone(state.textLibrary)
    });

    return page;
  }

  function aiDeckCompilePlan(plan){
    const usedImageIds = new Set();

    return plan.pages.map(pagePlan =>
      aiDeckCompilePage(pagePlan, usedImageIds)
    );
  }

  function aiDeckApplyContacts(plan){
    if (!plan.contacts || typeof plan.contacts !== "object") return;

    state.ui.siteLinks = Object.assign(
      {},
      state.ui.siteLinks || {},
      {
        email:plan.contacts.email || state.ui.siteLinks?.email || "",
        telegram:plan.contacts.telegram || state.ui.siteLinks?.telegram || "",
        website:plan.contacts.website || state.ui.siteLinks?.website || "",
        behance:plan.contacts.behance || state.ui.siteLinks?.behance || "",
        instagram:plan.contacts.instagram || state.ui.siteLinks?.instagram || ""
      }
    );

    if (plan.document?.title) {
      state.ui.siteTitle = String(plan.document.title);
    }
  }

  function aiDeckResetSelection(firstPage){
    // FIX_AI_EDIT_LOCK_V1: жёстко возвращаем редактор в чистое интерактивное состояние.
    state.ui.selectedPageId = firstPage?.id || null;
    state.ui.selectedBlockIds = [];
    state.ui.selectedBlockId = null;
    state.ui.selectedSlotId = null;
    state.ui.editingBlockId = null;
    state.ui.preview = false; // на случай, если редактор случайно остался в preview
    ensureStateShape(state); // финальная нормализация уже ГЛОБАЛЬНОГО state (безопасно, один раз)
    // FIX_AI_EDIT_LOCK_V2: принудительно сбрасываем кэш синхронизации меню,
    // чтобы applyUI() заново подхватил новые AI-страницы, а не старую активную.
    // Без этого возможен рассинхрон селекторов и "залипание" неактивного состояния UI.
    state.ui.lastSyncedPageId = null;
  }

  function aiDeckBuildNew(plan){
    const compiledPages = aiDeckCompilePlan(plan);

    pushHistory();

    state.pages = compiledPages;
    aiDeckApplyContacts(plan);

    if (plan.document?.visualStyle === "editorial-blue") {
      state.ui.activeDeck = "EDITORIAL_BLUE_0_1";
    }

    if (plan.document?.visualStyle === "cinematic") {
      state.ui.activeDeck = "CINEMATIC_PORTFOLIO";
    }

    aiDeckResetSelection(compiledPages[0]);

    renderAll();
    applyUI();
    aiDeckRenderProjectInfo();

    return compiledPages.length;
  }

  function aiDeckInsertAfterSelected(plan){
    const compiledPages = aiDeckCompilePlan(plan);

    pushHistory();

    const selectedIndex = state.ui.selectedPageId
      ? state.pages.findIndex(page => page.id === state.ui.selectedPageId)
      : -1;

    if (selectedIndex >= 0) {
      state.pages.splice(selectedIndex + 1, 0, ...compiledPages);
    } else {
      state.pages.push(...compiledPages);
    }

    aiDeckApplyContacts(plan);
    aiDeckResetSelection(compiledPages[0]);

    renderAll();
    applyUI();
    aiDeckRenderProjectInfo();

    return compiledPages.length;
  }

  function aiDeckUpdateExisting(plan){
    const compiledPages = aiDeckCompilePlan(plan);

    pushHistory();

    compiledPages.forEach((compiledPage, index) => {
      const existingPage = state.pages[index];

      if (existingPage) {
        compiledPage.id = existingPage.id;
        state.pages[index] = compiledPage;
      } else {
        state.pages.push(compiledPage);
      }
    });

    aiDeckApplyContacts(plan);
    aiDeckResetSelection(state.pages[0]);

    renderAll();
    applyUI();
    aiDeckRenderProjectInfo();

    return {
      updated:Math.min(compiledPages.length, state.pages.length),
      added:Math.max(0, compiledPages.length - state.pages.length)
    };
  }

  function aiDeckOpen(){
    if (!aiEls.overlay) return;

    aiEls.overlay.classList.add("open");
    aiEls.overlay.setAttribute("aria-hidden", "false");

    aiDeckRenderProjectInfo();

      aiEls.jsonInput?.focus();
    });

    if (!aiEls.overlay) return;

    aiEls.overlay.classList.remove("open");
    aiEls.overlay.setAttribute("aria-hidden", "true");
  }

    const result = aiDeckValidateAndPreview(actionName);

    if (!result?.validation?.valid) return null;

    return result.plan;
  }

    aiDeckSetStatus(
      `Update stopped.\n${error.message}`,
      "error"
    );
  }
});

        function aiCatalogDownload(fileName, data){
          const blob = new Blob(
            [JSON.stringify(data, null, 2)],
            { type:"application/json;charset=utf-8" }
          );

          const url = URL.createObjectURL(blob);
          const anchor = document.createElement("a");

          anchor.href = url;
          anchor.download = fileName;
          anchor.click();

          setTimeout(() => URL.revokeObjectURL(url), 15000);
        }

        function aiCatalogGetSemanticType(block){
          if (!block || block.kind !== "text") return null;

          const role = String(block.role || "").trim();

          if (AI_ROLE_SEMANTIC_TYPES[role]) {
            return AI_ROLE_SEMANTIC_TYPES[role];
          }

          return (
            block.style?.semanticType ||
            inferSemanticType(role, "", block.content || "")
          );
        }

        function aiCatalogGetTextCapacity(block){
          const width = Math.max(1, Number(block.w || 1));
          const height = Math.max(1, Number(block.h || 1));
          const fontSize = Math.max(10, Number(block.style?.fontSize || 16));
          const lineHeight = Math.max(0.8, Number(block.style?.lineHeight || 1.2));

          const lines = height / (fontSize * lineHeight);
          const charsPerLine = width / (fontSize * 0.48);
          const roughCapacity = lines * charsPerLine;

          if (roughCapacity <= 45) return "micro";
          if (roughCapacity <= 110) return "short";
          if (roughCapacity <= 280) return "medium";
          return "long";
        }

        function aiCatalogGetTextGuidance(role, capacity){
          const semanticType = AI_ROLE_SEMANTIC_TYPES[role] || "body";

          if (semanticType === "title" || semanticType === "overlayTitle") {

          }

          if (semanticType === "statement") {

          }

          if (semanticType === "subtitle") {
            return "Одна компактная поясняющая строка.";
          }

          if (semanticType === "meta" || semanticType === "smallMeta") {
            return "Короткий label, категория, services или metadata. Допустимы переносы строк.";
          }

          if (role === "steps") {
            return "Краткая последовательность этапов через переносы строк.";

          if (role === "services") {
            return "Структурированный список услуг, проекта или deliverables через переносы строк.";
          }

          if (role === "values" || role === "value1") {

          }

          if (capacity === "micro") {
            return "Очень короткий текст: 1–5 слов.";
          }

          if (capacity === "short") {

        }

        function aiCatalogGetTemplateImageCount(blocks){
          let count = 0;

          blocks.forEach(block => {
            if (block.kind === "image") count++;

            if (block.kind === "frame") {
              count += (block.slots || []).length || getFrameLayoutCount(block.layout || "2x2");
            }
          });

          return count;
        }

        function aiCatalogDescribeTemplate(template, customIds){
          const samplePage = createTemplatePage(template);
          const blocks = samplePage.blocks || [];

          const textSlots = blocks
            .filter(block => block.kind === "text")
            .map(block => {
              const role = String(block.role || "").trim();
              const semanticType = aiCatalogGetSemanticType(block);
              const capacity = aiCatalogGetTextCapacity(block);

              return {
                role,
                semanticType,
                capacity,
                description:AI_ROLE_DESCRIPTIONS[role] || "Текстовый блок.",
                guidance:aiCatalogGetTextGuidance(role, capacity)
              };
            });

          const imageSlots = blocks
            .filter(block => block.kind === "image")
            .map(block => {
              const isFullBleed = (
                Number(block.x) === 0 &&
                Number(block.y) === 0 &&
                Number(block.w) === PAGE_W &&
                Number(block.h) === PAGE_H
              );

              return {
                role:String(block.role || "").trim(),
                description:AI_ROLE_DESCRIPTIONS[block.role] || "Image slot.",
                isFullBleed
              };
            });

          const frameSlots = blocks
            .filter(block => block.kind === "frame")
            .map(block => ({
              role:String(block.role || "").trim(),
              description:AI_ROLE_DESCRIPTIONS[block.role] || "Frame block.",
              layout:block.layout || "2x2",
              slotCount:(block.slots || []).length || getFrameLayoutCount(block.layout || "2x2"),
              gap:Number(block.gap || 0),
              emphasis:block.emphasis || "medium",
              asymmetry:block.asymmetry || "medium"
            }));

          const totalImageSlots = aiCatalogGetTemplateImageCount(blocks);
          const hasFullBleedImage = imageSlots.some(slot => slot.isFullBleed);
          const hasOverlayText = blocks.some(block => {
            if (block.kind !== "text") return false;
            const type = aiCatalogGetSemanticType(block);
            return type === "overlayTitle" || type === "overlayBody";
          });

          return {
            id:template.id,
            label:template.label,
            templateGroup:template.templateGroup,
            visualStyle:template.visualStyle || "cinematic",
            source:customIds.has(template.id) ? "custom-js" : "core-presentation-html",

            usageHint:aiCatalogGetUsageHint(
              template.templateGroup,
              totalImageSlots
            ),

            layoutSummary:{
              textSlotCount:textSlots.length,
              imageSlotCount:imageSlots.length,
              frameBlockCount:frameSlots.length,
              totalImageSlots,
              hasFullBleedImage,
              hasOverlayText
            },

            textRoles:textSlots.map(slot => slot.role),
            textSlots,
            imageSlots,
            frameSlots
          };
        }

        function aiCatalogGetImagePoolManifest(){
          const actualGroups = new Map();

          (state.imagePool?.items || []).forEach(item => {
            const conceptIndex = normalizeConceptNumber(item.concept || 0);
            const conceptSubfolder = normalizeConceptSubfolder(item.subfolder || "root");

            if (conceptIndex <= 0) return;

            const key = `${conceptIndex}:${conceptSubfolder}`;

            if (!actualGroups.has(key)) {
              actualGroups.set(key, {
                conceptIndex,
                conceptSubfolder,
                imageCount:0
              });
            }

            actualGroups.get(key).imageCount++;
          });

          const groups = Array.from(actualGroups.values())
            .sort((a, b) => {
              if (a.conceptIndex !== b.conceptIndex) {
                return a.conceptIndex - b.conceptIndex;
              }

              return String(a.conceptSubfolder).localeCompare(
                String(b.conceptSubfolder),
                undefined,
                { numeric:true }
              );
            });

          const conceptsMap = new Map();

          groups.forEach(group => {
            if (!conceptsMap.has(group.conceptIndex)) {
              conceptsMap.set(group.conceptIndex, {
                conceptIndex:group.conceptIndex,
                totalImageCount:0,
                allowedConceptSubfolders:["all"],
                subfolders:[]
              });
            }

            const concept = conceptsMap.get(group.conceptIndex);

            concept.totalImageCount += group.imageCount;

            if (!concept.allowedConceptSubfolders.includes(group.conceptSubfolder)) {
              concept.allowedConceptSubfolders.push(group.conceptSubfolder);
            }

            concept.subfolders.push({
              conceptSubfolder:group.conceptSubfolder,
              imageCount:group.imageCount
            });
          });

          const concepts = Array.from(conceptsMap.values())
            .sort((a, b) => a.conceptIndex - b.conceptIndex)
            .map(concept => {
              concept.allowedConceptSubfolders = [
                "all",
                ...concept.allowedConceptSubfolders
                  .filter(value => value !== "all")
                  .sort((a, b) => String(a).localeCompare(String(b), undefined, { numeric:true }))
              ];

              return concept;
            });

          return {
            schemaVersion:"ai-image-pool-manifest-v2",
            generatedAt:new Date().toISOString(),

            purpose:[
              "Этот файл описывает доступность изображений по conceptIndex и conceptSubfolder.",
              "Он не содержит изображения и не содержит конкретные filenames.",
              "Нейросеть выбирает только conceptIndex и conceptSubfolder.",
              "Редактор сам распределяет реальные изображения по image slots."
            ],

            selectionRules:[
              "Используй только conceptIndex из concepts[].conceptIndex.",
              "Для каждого conceptIndex используй только conceptSubfolder из allowedConceptSubfolders.",
              "all — виртуальная агрегированная опция: использовать любые изображения внутри данного концепта.",
              "root — использовать только изображения из корня концепта.",
              "01–10 — использовать только соответствующую реальную подпапку.",
              "Если noRepeats равно true, totalImageSlots выбранного template не должен превышать imageCount выбранной группы.",
              "Для conceptSubfolder all используй totalImageCount соответствующего concept.",
              "Не указывай filenames и не пытайся выбрать конкретную картинку."
            ],

            concepts,
            groups
          };
        }

        function aiCatalogBuildStyleCatalog(styleId){
          const customIds = aiCatalogGetCustomTemplateIds();
          const style = AI_STYLE_PROFILES[styleId];

          const templates = Object.values(PAGE_TEMPLATE_LIBRARY)
            .flatMap(group => group.templates || [])
            .filter(template => (template.visualStyle || "cinematic") === styleId)
            .map(template => aiCatalogDescribeTemplate(template, customIds));

          const groupSummary = {};

          templates.forEach(template => {
            const groupName = template.templateGroup || "Other";

            if (!groupSummary[groupName]) {
              groupSummary[groupName] = 0;
            }

            groupSummary[groupName]++;
          });

          return {
            schemaVersion:"ai-template-catalog-v3",
            generatedAt:new Date().toISOString(),

            catalogPurpose:[
              "Технический источник истины для выбора templateId.",

            styleProfile:style,

              width:PAGE_W,
              height:PAGE_H
            },

            requiredRules:[
              "Используй только templateId из templates[].id.",
              "Заполняй только text roles, реально существующие в textSlots выбранного template.",

            groupSummary,
            templates
          };
        }

          );

            String(event.key || "").toLowerCase() === AI_EXPORT_SHORTCUT.key;

          if (!isShortcut) return;

          event.preventDefault();
          event.stopImmediatePropagation();

          aiCatalogExportAll();
        }, true);
      })();

        // Превью: картинки реальные, текст — реальными буквами (видно содержимое).
        function buildThumb(page){
          let html = '<div class="aipc-thumb" style="container-type:size;">';
          (page.blocks || []).forEach(b => {
            if (b.visible === false) return;
            const left = (b.x / 1123) * 100, top = (b.y / 794) * 100;
            const w = (b.w / 1123) * 100, h = (b.h / 794) * 100;
            const pos = 'left:'+left+'%;top:'+top+'%;width:'+w+'%;height:'+h+'%;';
            if (b.kind === "image") {
              const src = window.__aipcGetSrc ? window.__aipcGetSrc(b) : (b.src || "");
              if (src && !b.isTemplatePlaceholder) html += '<div class="aipc-tb" style="'+pos+'"><img src="'+esc(src)+'" loading="lazy"></div>';
              else html += '<div class="aipc-tb" style="'+pos+'background:#e7e5df;"></div>';
            } else if (b.kind === "frame") {
              html += '<div class="aipc-tb" style="'+pos+'background:#f0eee8;"></div>';
            } else if (b.kind === "shape") {
              const fill = (b.style && b.style.fill) ? b.style.fill : "#fff";
              html += '<div class="aipc-tb" style="'+pos+'background:'+esc(fill)+';"></div>';
            } else if (b.kind === "text") {
              const txt = window.__aipcHtmlToText ? window.__aipcHtmlToText(b.content||"") : (b.content||"");
              const fsPct = Math.max(2.4, ((b.style && b.style.fontSize) ? b.style.fontSize : 16) / 1123 * 100);
              const col = (b.style && b.style.color) ? b.style.color : "#111";
              const al = (b.style && b.style.textAlign) ? b.style.textAlign : "left";
              html += '<div class="aipc-tb aipc-tb-text" style="'+pos+'font-size:'+fsPct+'cqw;color:'+esc(col)+';text-align:'+esc(al)+';">'+esc(txt)+'</div>';
            }
          });
          html += '</div>';
          return html;
        }

        function buildMeta(page){
          const t=(page.blocks||[]).filter(b=>b.kind==="text").length;
          const i=(page.blocks||[]).filter(b=>b.kind==="image").length;
          const f=(page.blocks||[]).filter(b=>b.kind==="frame").length;
          return "templateId: "+(page.templateId||"—")+" · text:"+t+" · img:"+i+" · frame:"+f;

        function buildSnapshot(page){
          const blocks = (page.blocks||[]).map(b=>{
            const base={kind:b.kind,role:b.role||"",x:b.x,y:b.y,w:b.w,h:b.h};
            if(b.kind==="text"){ base.textType=(b.style&&b.style.semanticType)?b.style.semanticType:"body"; base.text=window.__aipcHtmlToText?window.__aipcHtmlToText(b.content||""):(b.content||""); }
            if(b.kind==="shape") base.fill=(b.style&&b.style.fill)?b.style.fill:"";
            if(b.kind==="image") base.hasImage=!!(window.__aipcGetSrc&&window.__aipcGetSrc(b)&&!b.isTemplatePlaceholder);
            if(b.kind==="frame"){ base.layout=b.layout||"2x2"; base.slotCount=(b.slots||[]).length; }
            return base;
          });
          return {
            templateId:page.templateId||"", templateGroup:page.templateGroup||"", label:page.sourceLabel||"",
            visualStyle:page.visualStyle||"cinematic", conceptIndex:page.conceptIndex||0, conceptSubfolder:page.conceptSubfolder||"all",
            counts:{ text:blocks.filter(b=>b.kind==="text").length, image:blocks.filter(b=>b.kind==="image").length, frame:blocks.filter(b=>b.kind==="frame").length },
            blocks
          };
        }

  function syncInspector() {
    const selection = document.getElementById("selectionToolbar");
    const parking = document.getElementById("approvedSelectionParking");
    const activeSection = document.body.dataset.approvedSplitSection || "";
    const frameSlot = document.getElementById("approvedSplitFrameSlot");
    const textSlot = document.getElementById("approvedSplitTextSlot");

    if (!selection || !parking || !frameSlot || !textSlot) return;

    frameSlot.querySelector(".approved-inspector-empty")?.remove();
    textSlot.querySelector(".approved-inspector-empty")?.remove();

    const kind = getInspectorKind();

        if (activeSection === "frame" && kind === "frame") {
      /*
        Не appendChild повторно, если toolbar уже находится
        в Frame Inspector. Повторный appendChild на mouseup
        закрывает открытый native <select>.

      if (selection.parentElement !== frameSlot) {
        frameSlot.appendChild(selection);
      }
      return;
    }

    if (activeSection === "text" && kind === "text") {
      /*
        Та же защита для Text Inspector.
      */
      if (selection.parentElement !== textSlot) {
        textSlot.appendChild(selection);
      }
      return;
    }

    parking.appendChild(selection);

    if (activeSection === "frame") {
      frameSlot.insertAdjacentHTML(
        "beforeend",
        `
          <div class="approved-inspector-empty">
            Select one Image, Frame or Shape block on the canvas.
          </div>
        `
      );
