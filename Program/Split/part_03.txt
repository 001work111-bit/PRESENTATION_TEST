      function formatBytesApprox(chars){
        const bytes = Math.round(chars * 0.75);
        if (bytes < 1024) return bytes + " B";
        if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + " KB";
        return (bytes / (1024 * 1024)).toFixed(1) + " MB";
      }

      function computeMemoryStats(){
        let imageBlocks = 0, frameSlots = 0, previewChars = 0, originalChars = 0;
        state.pages.forEach(page => {
          (page.blocks || []).forEach(b => {
            if (b.kind === "image") imageBlocks++;
            if (b.kind === "frame") frameSlots += (b.slots || []).length;
          });
        });
        (state.imagePool.items || []).forEach(it => {
          previewChars += (it.previewSrc || "").length;
          originalChars += (it.originalSrc || "").length;
        });
        const approxChars = previewChars + originalChars;
        let ratio = approxChars / (170 * 1024 * 1024);
        ratio += state.pages.length * 0.005;
        ratio += frameSlots * 0.002;
        ratio += imageBlocks * 0.002;
        ratio = clamp(ratio, 0, 1);
        let level = "low";
        let warning = "Load is low.";
        if (ratio >= 0.25) { level = "medium"; warning = "Project is getting heavier."; }
        if (ratio >= 0.5) { level = "high"; warning = "High load. Prefer smaller folders and previews."; }
        if (ratio >= 0.8) { level = "danger"; warning = "Danger zone. Browser may become unstable with many originals."; }
        state.ui.memoryStats = {
          pages:state.pages.length,
          imageBlocks,
          frameSlots,
          pool:state.imagePool.items.length,
          previewChars,
          originalChars,
          approxChars,
          ratio,
          level,
          warning
        };
      }

      function updateMemoryUI(){

        const m = state.ui.memoryStats;

          <div>Image Blocks: ${m.imageBlocks}</div>
          <div>Slots: ${m.frameSlots}</div>
          <div>Preview: ${formatBytesApprox(m.previewChars)}</div>
          <div>Original: ${formatBytesApprox(m.originalChars)}</div>
        `;
        els.memoryLoadBar.style.width = `${Math.round(m.ratio * 100)}%`;
        els.memoryStatsText.textContent = `Load: ${m.level} — ${m.warning}`;
      }

        const a = document.createElement("a");

        a.click();
        setTimeout(() => URL.revokeObjectURL(a.href), 10000);

      function importTextLibraryCsv(file){
        if (!file) return;
        const reader = new FileReader();
        reader.onload = () => {
          try {
            const rows = parseCsv(String(reader.result || ""));
            if (!rows.length || rows.length < 2) return toast("Text Library CSV empty");
            const header = rows[0].map(v => String(v || "").trim().toLowerCase());
            const gi = header.indexOf("group_name");
            const ti = header.indexOf("item_title");
            const xi = header.indexOf("item_text");
            if (gi === -1 || ti === -1 || xi === -1) return toast("Invalid Text Library CSV");
            pushHistory();
            const groupsMap = new Map((state.textLibrary.groups || []).map(group => [group.name, group]));
            rows.slice(1).forEach(row => {
              const groupName = String(row[gi] || "").trim();
              const itemTitle = String(row[ti] || "").trim();
              const itemText = String(row[xi] || "");
              if (!groupName && !itemTitle && !itemText) return;
              let group = groupsMap.get(groupName);
              if (!group) {
                group = { id:uid("tlg"), name:groupName || "New Group", items:[] };
                state.textLibrary.groups.push(group);
                groupsMap.set(group.name, group);
              }
              group.items.push({
                id:uid("tli"),
                title:itemTitle || "New Text",
                text:itemText || ""
              });
            });
            state.textLibrary = normalizeTextLibraryState(state.textLibrary);
            renderTextLibraryPanel();
            applyUI();
            toast("Text Library CSV imported");
          } catch (err) {
            console.error(err);
            toast("Text Library CSV import error");
          }
        };
        reader.readAsText(file, "utf-8");
      }

      function handleHotkeyCaptureKeydown(e){
        if (!hotkeyCapturingKey) return false;

        const key = e.key;

        if (key === "Escape") {
          e.preventDefault();
          e.stopPropagation();
          hotkeyCapturingKey = null;
          renderHotkeysPanel();
          return true;
        }

        const combo = hotkeyComboFromEvent(e);
        if (!combo) return true;

        e.preventDefault();
        e.stopPropagation();

        const target = hotkeyCapturingKey;
        hotkeyCapturingKey = null;
        setHotkey(target, combo);
        return true;
      }

      function getImageSize(url) {
        return new Promise(resolve => {
          const img = new Image();
          img.onload = () => resolve({ width: img.naturalWidth, height: img.naturalHeight });
          img.onerror = () => resolve(null);
          img.src = url;
        });
      }
      function generateResizedImageData(dataUrl, maxSide, quality){
        return new Promise(resolve => {
          const img = new Image();
          img.onload = () => {
            let w = img.naturalWidth || img.width;
            let h = img.naturalHeight || img.height;
            const scale = Math.min(1, maxSide / Math.max(w, h));
            const nw = Math.max(1, Math.round(w * scale));
            const nh = Math.max(1, Math.round(h * scale));
            const canvas = document.createElement("canvas");
            canvas.width = nw;
            canvas.height = nh;
            const ctx = canvas.getContext("2d", { alpha:false });
            if (ctx && "imageSmoothingEnabled" in ctx) {
              ctx.imageSmoothingEnabled = true;
              ctx.imageSmoothingQuality = "high";
            }
            ctx.drawImage(img, 0, 0, nw, nh);
            resolve({
              src: canvas.toDataURL("image/jpeg", quality),
              naturalSize:{ width:w, height:h }
            });
          };
          img.onerror = () => resolve({ src:dataUrl, naturalSize:null });
          img.src = dataUrl;
        });
      }

      function htmlToTextareaValue(html){
        const source = String(html || "");
        if (!source) return "";
        const normalized = source
          .replace(/\r\n/g, "\n")
          .replace(/<div><br><\/div>/gi, "\n")
          .replace(/<div>/gi, "\n")
          .replace(/<\/div>/gi, "")
          .replace(/<p>/gi, "\n")
          .replace(/<\/p>/gi, "")
          .replace(/<br\s*\/?>/gi, "\n");
        const div = document.createElement("div");
        div.innerHTML = normalized;
        return (div.textContent || div.innerText || "").replace(/^\n+/, "");
      }

      function textareaValueToHtml(value){
        const escaped = String(value || "")
          .replace(/\r\n/g, "\n")
          .replace(/\r/g, "\n")
          .replace(/&/g, "&amp;")
          .replace(/</g, "&lt;")
          .replace(/>/g, "&gt;");
        return escaped.replace(/\n/g, "<br>");
      }

          .replace(/<div>/gi, "")

      }

      function stripHtml(html){
        const div = document.createElement("div");

        const area = document.createElement("textarea");

      function isEditingTextTarget(target){
        if (!target) return false;
        // Реальное редактирование текста: contentEditable или textarea.
        if (target.isContentEditable) return true;
        if (target.tagName === "TEXTAREA") return true;
        // INPUT блокирует хоткеи ТОЛЬКО если это текстовое поле, куда реально печатают.
        if (target.tagName === "INPUT") {
          const type = String(target.type || "text").toLowerCase();
          const textTypes = ["text","search","email","url","tel","password","number"];
          return textTypes.includes(type);
        }
        // SELECT и BUTTON и прочее НЕ считаем вводом текста —
        // иначе хоткеи "залипают" после клика по кнопке/селекту (Ctrl+D, Template Select и т.п.).
        return false;
      }

      function getImageRenderData(obj){
        if (!obj) return { src:"" };
        if (obj.poolImageId) {

          return { src:"", previewSrc:"", originalSrc:"", missingPoolRef:true };
        }
        return { src:obj.src || "", previewSrc:obj.src || "", originalSrc:obj.src || "" };
      }

      function applyImageTransformToElement(imgEl, block, slotRect){
        ensureImageTransform(block);
        const t = block.imageTransform;
        const scale = Math.max(1, t.scale || 1);
        const bw = slotRect ? slotRect.w : block.w;
        const bh = slotRect ? slotRect.h : block.h;
        if (!block.naturalSize) {
          imgEl.style.width = `${bw * scale}px`;
          imgEl.style.height = `${bh * scale}px`;
          imgEl.style.transform = `translate(calc(-50% + ${t.offsetX}px), calc(-50% + ${t.offsetY}px))`;
          return;
        }
        const { width:nw, height:nh } = block.naturalSize;
        const baseScale = Math.max(bw / nw, bh / nh);
        const finalScale = baseScale * scale;
        imgEl.style.width = `${nw * finalScale}px`;
        imgEl.style.height = `${nh * finalScale}px`;
        imgEl.style.transform = `translate(calc(-50% + ${t.offsetX}px), calc(-50% + ${t.offsetY}px))`;
      }

      function constrainImageTransform(block, slotRect){
        ensureImageTransform(block);
        const t = block.imageTransform;
        const scale = Math.max(1, t.scale || 1);
        const bw = slotRect ? slotRect.w : block.w;
        const bh = slotRect ? slotRect.h : block.h;
        let extraW, extraH;
        if (block.naturalSize) {
          const baseScale = Math.max(bw / block.naturalSize.width, bh / block.naturalSize.height);
          const finalW = block.naturalSize.width * baseScale * scale;
          const finalH = block.naturalSize.height * baseScale * scale;
          extraW = Math.max(0, (finalW - bw) / 2);
          extraH = Math.max(0, (finalH - bh) / 2);
        } else {
          extraW = (bw * scale - bw) / 2;
          extraH = (bh * scale - bh) / 2;
        }
        t.offsetX = clamp(t.offsetX, -extraW, extraW);
        t.offsetY = clamp(t.offsetY, -extraH, extraH);
      }

      function emphasisFactor(v){

      function getCustomLayoutNormalizedRects(block){
        const v = block.layoutVariant || 0;
        const e = emphasisFactor(block.emphasis || "medium");
        const a = asymmetryFactor(block.asymmetry || "medium");
        if (block.layout === "bento-4-a") {
          const left = clamp(0.58 * e, 0.5, 0.72);
          const top = clamp(0.58 * a, 0.46, 0.7);

            ? [
                {x:0,y:0,w:left,h:top},

                {x:left,y:(top*0.52),w:1-left,h:1-(top*0.52)},
                {x:0,y:top,w:left,h:1-top}
              ]
            : [
                {x:0,y:0,w:left,h:(top*0.52)},
                {x:0,y:(top*0.52),w:left,h:1-(top*0.52)},
                {x:left,y:0,w:1-left,h:top},
                {x:left,y:top,w:1-left,h:1-top}
              ];
        }
        if (block.layout === "bento-4-b") {
          const right = clamp(0.42 / e, 0.3, 0.5);
          const left = 1 - right;
          const bottomLeft = clamp(0.38 * a, 0.28, 0.48);
          return v % 2 === 0
            ? [
                {x:0,y:0,w:left,h:1-bottomLeft},
                {x:0,y:1-bottomLeft,w:left*0.54,h:bottomLeft},
                {x:left*0.54,y:1-bottomLeft,w:left*0.46,h:bottomLeft},
                {x:left,y:0,w:right,h:1}
              ]
            : [
                {x:0,y:0,w:right,h:1},
                {x:right,y:0,w:1-right,h:1-bottomLeft},
                {x:right,y:1-bottomLeft,w:(1-right)*0.46,h:bottomLeft},
                {x:right+(1-right)*0.46,y:1-bottomLeft,w:(1-right)*0.54,h:bottomLeft}
              ];
        }
        if (block.layout === "bento-4-c") {
          const top = clamp(0.44 / e + 0.06, 0.34, 0.56);
          const leftTop = clamp(0.64 * a, 0.54, 0.74);
          const bottomMain = clamp(0.54 * e, 0.46, 0.68);
          return v % 2 === 0
            ? [
                {x:0,y:0,w:leftTop,h:top},
                {x:leftTop,y:0,w:1-leftTop,h:top},
                {x:0,y:top,w:bottomMain,h:1-top},

              ]
            : [
                {x:0,y:0,w:1-leftTop,h:top},
                {x:1-leftTop,y:0,w:leftTop,h:top},
                {x:0,y:top,w:1-bottomMain,h:1-top},
                {x:1-bottomMain,y:top,w:bottomMain,h:1-top}
              ];
        }
        if (block.layout === "mosaic-4-a") {
          const splitX = clamp(0.52 * e, 0.42, 0.64);
          const splitY = clamp(0.54 * a, 0.44, 0.66);
          return v % 2 === 0
            ? [
                {x:0,y:0,w:splitX,h:splitY},
                {x:splitX,y:0,w:1-splitX,h:splitY*0.64},
                {x:splitX,y:splitY*0.64,w:1-splitX,h:1-(splitY*0.64)},
                {x:0,y:splitY,w:splitX,h:1-splitY}
              ]
            : [
                {x:0,y:0,w:1-splitX,h:splitY*0.64},
                {x:0,y:splitY*0.64,w:1-splitX,h:1-(splitY*0.64)},
                {x:1-splitX,y:0,w:splitX,h:splitY},
                {x:1-splitX,y:splitY,w:splitX,h:1-splitY}
              ];
        }
        if (block.layout === "mosaic-4-b") {
          const top = clamp(0.34 + (e-1)*0.18, 0.28, 0.48);
          const lowerLeft = clamp(0.36 + (a-1)*0.24, 0.26, 0.48);
          const lowerCenter = clamp(0.32, 0.24, 0.38);
          const last = Math.max(0.16, 1 - lowerLeft - lowerCenter);
          return v % 2 === 0
            ? [
                {x:0,y:0,w:1,h:top},
                {x:0,y:top,w:lowerLeft,h:1-top},
                {x:lowerLeft,y:top,w:lowerCenter,h:1-top},
                {x:lowerLeft+lowerCenter,y:top,w:last,h:1-top}
              ]
            : [
                {x:0,y:0,w:1,h:top},
                {x:0,y:top,w:last,h:1-top},
                {x:last,y:top,w:lowerCenter,h:1-top},
                {x:last+lowerCenter,y:top,w:lowerLeft,h:1-top}
              ];
        }
        return null;
      }

      function normalizedRectsToActual(block, normRects){
        const gap = Math.max(0, block.gap || 0);
        return normRects.map((nr, i) => {
          const rightGaps = nr.x > 0 ? 1 : 0;
          const leftGaps = nr.x + nr.w < 1 ? 1 : 0;
          const topGaps = nr.y > 0 ? 1 : 0;
          const bottomGaps = nr.y + nr.h < 1 ? 1 : 0;
          const x = nr.x * block.w + rightGaps * gap;
          const y = nr.y * block.h + topGaps * gap;
          const w = nr.w * block.w - (rightGaps + leftGaps) * gap;
          const h = nr.h * block.h - (topGaps + bottomGaps) * gap;
          return { x, y, w, h, index:i };
        });
      }

      function getFrameSlotRects(block){
        const custom = getCustomLayoutNormalizedRects(block);
        if (custom) return normalizedRectsToActual(block, custom);
        const rc = FRAME_LAYOUTS[block.layout] || [2,2];
        const rows = rc[0], cols = rc[1], gap = Math.max(0, block.gap || 0);
        const cellW = (block.w - gap * (cols - 1)) / cols;
        const cellH = (block.h - gap * (rows - 1)) / rows;
        const rects = [];
        let i = 0;
        for (let r=0;r<rows;r++){
          for (let c=0;c<cols;c++){
            rects.push({
              x: c * (cellW + gap),
              y: r * (cellH + gap),
              w: cellW,
              h: cellH,
              index:i++
            });
          }
        }
        return rects;
      }

        if (!input) return;

          (page.blocks || []).forEach(block => {
            if (block.kind !== "text") return;

  function setTemplateFilterButtonState(button, isActive, isEnabled = true){
    if (!button) return;

    button.style.cursor = "pointer";

        Спокойный тёмно-синий active-state:

        "rgba(137, 174, 224, .62)",
        "important"
      );
      button.style.setProperty(
        "color",
        "#edf4ff",
        "important"
      );
    } else {
      button.style.removeProperty("background");
      button.style.removeProperty("border-color");
      button.style.removeProperty("color");
    }
  }

  function getFilteredCustomTemplates(
    allTemplates,
    currentStyle,
    currentCategory
  ) {
    if (!Array.isArray(allTemplates)) {
      return [];

    const favoritesOnly =
      !!window.TL_FILTER_FAV_ONLY;

      favoritesOnly &&
      !!window.TL_FILTER_ALL_CONCEPTS;

      const isFavorite =
        getEffectiveTemplateFavorite(item);
      /*
        Category никогда не отключается All Styles.
      */
      if (
        currentCategory &&
        currentCategory !== "all" &&
        item.templateGroup &&
        item.templateGroup.toLowerCase() !==
          currentCategory.toLowerCase()
      ) {
        return false;
      }

      if (favoritesOnly && !isFavorite) {
        return false;
      }

      /*
        Favorites + All Styles:
        оставляем category, снимаем только style filter.
      */
      if (allStylesForFavorites) {
        return true;
      }

      if (
        currentStyle &&
        currentStyle !== "all"
      ) {
        const itemStyle =
          String(item.visualStyle || "")
            .toLowerCase();

        const targetStyle =
          String(currentStyle || "")
            .toLowerCase()
            .replace(" only", "")
            .trim();

        if (
          itemStyle &&
          targetStyle &&
          !itemStyle.includes(targetStyle) &&
          !targetStyle.includes(itemStyle)
        ) {
          return false;
        }
      }

      return true;
    });

      function insertPageAfterSelected(page){
        const idx = state.ui.selectedPageId ? state.pages.findIndex(p => p.id === state.ui.selectedPageId) : -1;
        if (idx >= 0) state.pages.splice(idx + 1, 0, page);

      function deletePage(pId = state.ui.selectedPageId){
        if (!state.pages.length) return;
        const index = state.pages.findIndex(p => p.id === pId);
        if (index === -1) return;
        pushHistory();
        state.pages.splice(index, 1);
        const nextPage = state.pages[index] || state.pages[index - 1] || null;
        state.ui.selectedPageId = nextPage ? nextPage.id : null;
        state.ui.selectedBlockIds = [];
        state.ui.selectedBlockId = null;
        state.ui.selectedSlotId = null;
        state.ui.editingBlockId = null;
        renderAll();
        applyUI();
        toast("Page deleted");
      }

      function movePageUp(pId = state.ui.selectedPageId){
        const i = state.pages.findIndex(p => p.id === pId);
        if (i <= 0) return;
        pushHistory();
        [state.pages[i], state.pages[i - 1]] = [state.pages[i - 1], state.pages[i]];
        state.ui.selectedPageId = state.pages[i - 1].id;
        state.ui.selectedBlockIds = [];

      function placeCaretAtPoint(x, y, editableEl){
        editableEl.focus();

          const range = document.caretRangeFromPoint(x, y);
          if (range) {
            const sel = window.getSelection();
            sel.removeAllRanges();
            sel.addRange(range);
          }
        } else if (document.caretPositionFromPoint) {
          const pos = document.caretPositionFromPoint(x, y);
          if (pos) {
            const range = document.createRange();
            range.setStart(pos.offsetNode, pos.offset);
            range.collapse(true);
            const sel = window.getSelection();
            sel.removeAllRanges();
            sel.addRange(range);
          }
        }

      function onMouseMove(e){
        if (!dragState) return;
        const dx = e.clientX - dragState.startX;
        const dy = e.clientY - dragState.startY;

        if (dragState.type === "move") {
          e.preventDefault();
          if (Math.sqrt(dx*dx + dy*dy) > DRAG_THRESHOLD) dragState.moved = true;
          dragState.initialPositions.forEach((startPos, blockId) => {
            const block = dragState.page.blocks.find(b => b.id === blockId);
            if (!block) return;
            block.x = clamp(snap(startPos.x + dx), 0, PAGE_W - block.w);
            block.y = clamp(snap(startPos.y + dy), 0, PAGE_H - block.h);
          });
          renderAll();
          return;
        }

        if (dragState.type === "resize") {
          e.preventDefault();
          dragState.moved = true;
          const b = dragState.block;
          const minW = 40, minH = 20;
          let newX = b.x, newY = b.y, newW = b.w, newH = b.h;

          if (dragState.dir.includes("e")) newW = snap(Math.max(minW, dragState.startW + dx));
          if (dragState.dir.includes("s")) newH = snap(Math.max(minH, dragState.startH + dy));
          if (dragState.dir.includes("w")) {
            newW = snap(Math.max(minW, dragState.startW - dx));
            newX = snap(dragState.startXPos + (dragState.startW - newW));
          }
          if (dragState.dir.includes("n")) {
            newH = snap(Math.max(minH, dragState.startH - dy));
            newY = snap(dragState.startYPos + (dragState.startH - newH));
          }

          b.x = clamp(newX, 0, PAGE_W - newW);
          b.y = clamp(newY, 0, PAGE_H - newH);
          b.w = newW;
          b.h = newH;
          renderAll();
          return;
        }

        if (dragState.type === "selectBox") {
          const box = $("selectionBox");
          if (!box) return;
          const x = Math.min(dragState.startX, e.clientX);
          const y = Math.min(dragState.startY, e.clientY);
          box.style.cssText = `left:${x}px;top:${y}px;width:${Math.abs(dragState.startX - e.clientX)}px;height:${Math.abs(dragState.startY - e.clientY)}px;`;
          return;
        }

        if (dragState.type === "image-pan") {
          e.preventDefault();
          const t = dragState.target;
          ensureImageTransform(t);
          t.imageTransform.offsetX = dragState.startOffsetX + dx;
          t.imageTransform.offsetY = dragState.startOffsetY + dy;
          constrainImageTransform(t, dragState.slotRect);
          const sel = dragState.target.id && dragState.block.kind === "frame"
            ? document.querySelector(`.frame-slot[data-slot-id="${dragState.target.id}"] img`)
            : document.querySelector(`.block[data-id="${dragState.block.id}"] .image-frame img`);
          if (sel) applyImageTransformToElement(sel, t, dragState.slotRect);
        }
      }

/*
  Единый filter engine для всех templates:

  - core templates из index.html;
  - подключаемых templates из /templates,
    уже добавленных в PAGE_TEMPLATE_LIBRARY.

  Правила:
  1. Категория всегда остаётся текущей.
  2. Обычный режим использует Style Filter.
  3. Favorites оставляет только isFavorite/favorite templates.
  4. Favorites + All Styles снимает только фильтр стиля.
  5. All Styles никогда не смешивает разные категории.
*/
/*
  TEMPLATE FAVORITES REGISTRY V1
  Shared registry for template filters and Favorite: On / Off button.

  Priority:
  1. localStorage override;
  2. isFavorite / favorite in template metadata;
  3. false.
*/
const TEMPLATE_FAVORITES_STORAGE_KEY =
  "portfolio_editor_template_favorite_overrides_v1";

function getTemplateFavoriteOverrides(){
  try {
    const raw = localStorage.getItem(
      TEMPLATE_FAVORITES_STORAGE_KEY
    );

    const parsed = raw ? JSON.parse(raw) : {};

    return parsed &&
      typeof parsed === "object" &&
      !Array.isArray(parsed)
      ? parsed
      : {};
  } catch (error) {
    console.warn(
      "Could not read template favorite overrides",
      error
    );

    return {};
  }
}

function saveTemplateFavoriteOverrides(overrides){
  try {
    localStorage.setItem(
      TEMPLATE_FAVORITES_STORAGE_KEY,
      JSON.stringify(overrides || {})
    );
  } catch (error) {
    console.warn(
      "Could not save template favorite overrides",
      error
    );
  }
}

function getEffectiveTemplateFavorite(template){
  if (!template || !template.id) {
    return false;
  }

  const overrides = getTemplateFavoriteOverrides();

  if (
    Object.prototype.hasOwnProperty.call(
      overrides,
      template.id
    )
  ) {
    return overrides[template.id] === true;
  }

  return (
    template.isFavorite === true ||
    template.favorite === true
  );
}

function setTemplateFavoriteOverride(templateId, isFavorite){
  if (!templateId) {
    return;
  }

  const overrides = getTemplateFavoriteOverrides();

  overrides[templateId] = isFavorite === true;

  saveTemplateFavoriteOverrides(overrides);
}
function getFilteredTemplatesForCurrentCategory(
  categoryKey = state.ui.templateCategory
) {
  const group = PAGE_TEMPLATE_LIBRARY[categoryKey];

  if (!group || !Array.isArray(group.templates)) {
    return [];
  }

  const filterMode = $("templateStyleFilterSelect")
    ? $("templateStyleFilterSelect").value
    : (state.ui.templateStyleFilter || "auto");

  const currentStyle = getCurrentActiveStyle();

  const favoritesOnly = !!window.TL_FILTER_FAV_ONLY;

  /*
    Старое техническое имя остаётся для совместимости
    со старым UI и существующими handlers.
    В интерфейсе кнопка называется All Styles.
  */
  const allStylesForFavorites =
    favoritesOnly &&
    !!window.TL_FILTER_ALL_CONCEPTS;

  return group.templates.filter(template => {
    const isFavorite =
      getEffectiveTemplateFavorite(template);
    /*
      Favorites: только отмеченные templates.
    */
    if (favoritesOnly && !isFavorite) {
      return false;
    }

    /*
      Favorites + All Styles:
      category уже сохранена через group.templates,
      поэтому снимается только style filter.
    */
    if (allStylesForFavorites) {
      return true;
    }

    /*
      Обычный Style Filter.
    */
    if (filterMode === "all") {
      return true;
    }

    if (filterMode === "cinematic") {
      return template.visualStyle === "cinematic";
    }

    if (filterMode === "editorial-blue") {
      return template.visualStyle === "editorial-blue";
    }

    /*
      auto: текущий style выбранной страницы
      либо текущего активного deck.
    */
    return template.visualStyle === currentStyle;
  });
}

      function filteredPoolForPage(page){
        const all = Array.isArray(state.imagePool.items) ? state.imagePool.items : [];
        if (!all.length) return [];
        const c = Math.max(0, parseInt(page?.conceptIndex, 10) || 0);
        const subfolder = normalizeConceptSubfolder(page?.conceptSubfolder || "all");
        if (c === 0) return all.slice();
        if (subfolder === "all") {
          return all.filter(it => normalizeConceptNumber(it.concept) === c);
        }
        if (subfolder === "root") {
          return all.filter(it => normalizeConceptNumber(it.concept) === c && String(it.subfolder || "root") === "root");
        }
        return all.filter(it => normalizeConceptNumber(it.concept) === c && String(it.subfolder || "root") === subfolder);
      }

      function uniquePickFromPool(pool, used, options={}){
        if (!Array.isArray(pool) || !pool.length) return null;
        const strictNoRepeats = !!options.strictNoRepeats;
        if (strictNoRepeats) {
          const free = pool.filter(it => !used.has(it.id));
          if (!free.length) return null;
          const item = free[Math.floor(Math.random() * free.length)];
          if (item) used.add(item.id);
          return item || null;
        }
        const item = pool[Math.floor(Math.random() * pool.length)];
        if (item) used.add(item.id);
        return item || null;
      }

      function getAllRandomTargets(page, mode){
        const targets = [];
        if (!page) return targets;
        if (mode === "page") {
          page.blocks.forEach(b => {
            if (b.kind === "image") targets.push({ block:b, target:b });
            if (b.kind === "frame") (b.slots || []).forEach(s => targets.push({ block:b, slot:s, target:s }));
          });
        } else if (mode === "one") {
          const ctx = getSelectedImageContext();
          if (ctx) {
            if (ctx.type === "image") targets.push({ block:ctx.block, target:ctx.block });
            else if (ctx.slot) targets.push({ block:ctx.block, slot:ctx.slot, target:ctx.slot });
          }
        }
        return targets;
      }

      function randomizeTargets(page, targets, options={}){
        const pool = filteredPoolForPage(page);
        if (!targets.length) return { changed:false, count:0, emptyConcept:!pool.length, poolCount:pool.length };
        if (!pool.length) {
          if (options.clearUnassignedOnNoPool) {
            let cleared = 0;
            targets.forEach(t => {
              if (t.target.randomFixed) return;
              if (options.onlyEmpty && hasRealAssignedImage(t.target)) return;
              if (options.respectOnlyPlaceholder && !t.target.isTemplatePlaceholder && hasRealAssignedImage(t.target)) return;
              clearImageLikeAssignment(t.target);
              cleared++;
            });
            return { changed:cleared > 0, count:0, cleared, emptyConcept:true, poolCount:0 };
          }
          return { changed:false, count:0, emptyConcept:true, poolCount:0 };
        }

        const used = options.usedGlobal || new Set();
        const strictNoRepeats = options.strictNoRepeats != null ? !!options.strictNoRepeats : !!state.ui.noRepeats;
        let changed = false, count = 0;

        targets.forEach(t => {
          if (t.target.randomFixed) return;
          if (options.onlyEmpty && hasRealAssignedImage(t.target)) return;
          if (options.respectOnlyPlaceholder && !t.target.isTemplatePlaceholder && hasRealAssignedImage(t.target)) return;
          const item = uniquePickFromPool(pool, used, { strictNoRepeats });
          if (!item) {
            if (options.leaveUnfilledOnExhausted) clearImageLikeAssignment(t.target);
            return;
          }
          assignPoolImageToImageLike(t.target, item);
          changed = true;
          count++;
        });

        return { changed, count, emptyConcept:false, poolCount:pool.length };
      }

      function rerollAllUnlockedAcrossDocument(){
        if (!(state.imagePool.items || []).length) return toast("Pool empty");
        pushHistory();
        let changedPages = 0, rerolled = 0;
        let fallbackPages = 0;

        state.pages.forEach(page => {
          const targets = [];
          (page.blocks || []).forEach(b => {
            if (b.kind === "image" && !b.randomFixed) {
              targets.push({ block:b, target:b });
            }
            if (b.kind === "frame") {
              (b.slots || []).forEach(s => {
                if (!s.randomFixed) targets.push({ block:b, slot:s, target:s });
              });
            }
          });
          if (!targets.length) return;

          // FIX_RANDOM_ALL_V3:
          // used-набор ИНДИВИДУАЛЬНЫЙ на каждую страницу — ровно так же,
          // как работает Random Page. Это чинит ситуацию, когда общий used
          // на весь документ исчерпывал пул и страницы оставались пустыми.
          const usedPerPage = new Set();

          // Fallback: если concept > 0, но картинок этого концепта нет — берём весь пул.
          const prevConcept = page.conceptIndex;
          const prevSubfolder = page.conceptSubfolder;
          const conceptSet = (parseInt(page.conceptIndex, 10) || 0) > 0;
          const conceptEmpty = conceptSet && !filteredPoolForPage(page).length;

          if (conceptEmpty) {
            page.conceptIndex = 0;
            page.conceptSubfolder = "all";
            fallbackPages++;
          }

          const res = randomizeTargets(page, targets, {
            usedGlobal: usedPerPage,
            strictNoRepeats: !!state.ui.noRepeats,
            leaveUnfilledOnExhausted: false
          });

          // Возвращаем исходный concept страницы — метаданные не меняются.
          page.conceptIndex = prevConcept;
          page.conceptSubfolder = prevSubfolder;

          if (res.changed) changedPages++;
          rerolled += res.count;
        });

        if (rerolled > 0) {
          renderAll();
          applyUI();
          const note = fallbackPages > 0
            ? ` (страниц без своего концепта — из общего пула: ${fallbackPages})`
            : "";
          toast(`Random All: rerolled ${rerolled} items on ${changedPages} pages${note}`);
        } else {
          history.pop();
          historyIndex = history.length - 1;
          toast("Nothing to reroll");
        }
      }

        // Поднимаемся от папки файла на количество сегментов,

        // 2 подъёма = C:\ImagePool
        let rootPath = nodePath.dirname(firstFile.path);

          rootPath = nodePath.dirname(rootPath);
        }

        state.imagePool.rootPath = rootPath;
      }
